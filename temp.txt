function addOrReplaceTimestamp(tokens) {
  if (tsSpan) tsSpan.remove();
  tsSpan = document.createElement('span');
  tsSpan.className = 'msg-ts';
  const timeStr = nowTs();

  if (tokens != null) {
    tsSpan.innerHTML =
      `<strong style="color:var(--text-primary);font-weight:600;">${tokens.toLocaleString()} tokens</strong>` +
      `<span style="color:var(--text-muted);margin-left:4px;">| ${timeStr}</span>`;
  } else {
    tsSpan.textContent = timeStr;
  }

  bw.appendChild(tsSpan);
}
