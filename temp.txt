import google.generativeai as genai

genai.configure(api_key="YOUR_API_KEY")

model = genai.GenerativeModel("gemini-2.0-flash")

# Enable Google Search grounding
response = model.generate_content(
    "What is today's top news headline?",
    tools=[{"google_search": {}}]
)

# Check if grounding was used
candidate = response.candidates[0]
if hasattr(candidate, "grounding_metadata") and candidate.grounding_metadata:
    print("Web search (grounding) is WORKING")
    print("Search queries used:", candidate.grounding_metadata.search_queries)
    print("Sources:", [c.web.uri for c in candidate.grounding_metadata.grounding_chunks if c.web])
else:
    print("Web search did NOT trigger (or is not available)")
    print("Response:", response.text)
