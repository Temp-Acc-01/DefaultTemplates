// ─── Back to dashboard ───────────────────────────────────────────
const DASHBOARD_URL = 'http://YOUR_IP:YOUR_PORT';  // ← replace with your dashboard route

function confirmBackToDashboard() {
  document.getElementById('back-confirm-overlay').classList.add('open');
  document.body.style.overflow = 'hidden';
}

function closeBackConfirm() {
  document.getElementById('back-confirm-overlay').classList.remove('open');
  document.body.style.overflow = '';
}

function goToDashboard() {
  closeBackConfirm();
  window.location.href = DASHBOARD_URL;
}

// Close on overlay backdrop click
document.getElementById('back-confirm-overlay')?.addEventListener('click', function(e) {
  if (e.target === this) closeBackConfirm();
});
