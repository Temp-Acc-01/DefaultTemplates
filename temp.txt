# migration_add_id_and_wallets.py
"""
One-time migration to:
1. Add id column to nlq_user_details
2. Create nlq_token_wallets table
3. Create nlq_token_transactions table
4. Migrate existing api_tokens data to wallets
"""

import pymysql
import os
from dotenv import load_dotenv

load_dotenv()

MYSQL_HOST = os.environ.get("AUTH_MYSQL_HOST", "localhost")
MYSQL_PORT = int(os.environ.get("AUTH_MYSQL_PORT", 3306))
MYSQL_USER = os.environ.get("AUTH_MYSQL_USER", "nabh")
MYSQL_PASS = os.environ.get("AUTH_MYSQL_PASS", "nabh@2026")
MYSQL_DB   = os.environ.get("AUTH_MYSQL_DB", "NABH")

def migrate():
    conn = pymysql.connect(
        host=MYSQL_HOST,
        port=MYSQL_PORT,
        user=MYSQL_USER,
        password=MYSQL_PASS,
        database=MYSQL_DB,
        autocommit=False
    )
    
    try:
        with conn.cursor() as cur:
            # 1. Add id column to nlq_user_details
            print("Adding id column to nlq_user_details...")
            cur.execute("""
                ALTER TABLE nlq_user_details 
                ADD COLUMN id BIGINT AUTO_INCREMENT PRIMARY KEY FIRST
            """)
            
            # 2. Create token_wallets table
            print("Creating nlq_token_wallets table...")
            cur.execute("""
                CREATE TABLE IF NOT EXISTS nlq_token_wallets (
                    user_id BIGINT PRIMARY KEY,
                    total_allocated BIGINT NOT NULL DEFAULT 0,
                    total_used BIGINT NOT NULL DEFAULT 0,
                    available_tokens BIGINT NOT NULL DEFAULT 0,
                    updated_at TIMESTAMP DEFAULT NOW(),
                    FOREIGN KEY (user_id) REFERENCES nlq_user_details(id) ON DELETE CASCADE
                ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
            """)
            
            # 3. Create token_transactions table
            print("Creating nlq_token_transactions table...")
            cur.execute("""
                CREATE TABLE IF NOT EXISTS nlq_token_transactions (
                    id BIGINT AUTO_INCREMENT PRIMARY KEY,
                    user_id BIGINT NOT NULL,
                    type VARCHAR(20) NOT NULL,
                    tokens BIGINT NOT NULL,
                    request_id VARCHAR(100),
                    model_name VARCHAR(100),
                    balance_before BIGINT,
                    balance_after BIGINT,
                    metadata JSON,
                    created_at TIMESTAMP DEFAULT NOW(),
                    FOREIGN KEY (user_id) REFERENCES nlq_user_details(id) ON DELETE CASCADE,
                    INDEX idx_user_id (user_id),
                    INDEX idx_created_at (created_at),
                    INDEX idx_request_id (request_id)
                ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
            """)
            
            # 4. Migrate existing api_tokens data to wallets
            print("Migrating existing token data...")
            cur.execute("""
                SELECT id, username, api_tokens 
                FROM nlq_user_details 
                WHERE api_tokens IS NOT NULL
            """)
            users = cur.fetchall()
            
            migrated = 0
            for user in users:
                user_id = user['id']
                api_tokens = user['api_tokens']
                
                if api_tokens:
                    # Parse "available/total" format
                    if '/' in str(api_tokens):
                        parts = str(api_tokens).split('/')
                        available = int(parts[0])
                        total = int(parts[1])
                    else:
                        # Plain integer
                        available = int(api_tokens)
                        total = int(api_tokens)
                    
                    # Insert into wallet
                    cur.execute("""
                        INSERT INTO nlq_token_wallets 
                        (user_id, total_allocated, total_used, available_tokens, updated_at)
                        VALUES (%s, %s, 0, %s, NOW())
                        ON DUPLICATE KEY UPDATE
                            total_allocated = VALUES(total_allocated),
                            available_tokens = VALUES(available_tokens),
                            updated_at = NOW()
                    """, (user_id, total, available))
                    
                    # Log allocation transaction
                    cur.execute("""
                        INSERT INTO nlq_token_transactions
                        (user_id, type, tokens, balance_before, balance_after, metadata)
                        VALUES (%s, 'ALLOCATE', %s, 0, %s, JSON_OBJECT('migrated', true, 'note', 'Initial migration'))
                    """, (user_id, total, available))
                    
                    migrated += 1
            
            print(f"Migrated {migrated} users to wallet system")
            
            # 5. Optional: Drop old api_tokens column after verification
            # Uncomment after verifying migration:
            # print("Dropping old api_tokens column...")
            # cur.execute("ALTER TABLE nlq_user_details DROP COLUMN api_tokens")
            
        conn.commit()
        print("✅ Migration completed successfully!")
        
    except Exception as e:
        conn.rollback()
        print(f"❌ Migration failed: {e}")
        raise
    finally:
        conn.close()

if __name__ == "__main__":
    migrate()
