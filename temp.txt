function updateResponsiveLabels() {

  const tokenLabel = document.querySelector('.token-pill span');

  if (!tokenLabel) return;

  if (window.innerWidth <= 700) {
    tokenLabel.textContent = 'Tokens';
  } else {
    tokenLabel.textContent = 'Available Tokens';
  }
}

window.addEventListener('resize', updateResponsiveLabels);

setTimeout(updateResponsiveLabels, 100);
