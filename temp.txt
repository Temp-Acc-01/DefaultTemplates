## Chart Layout Rules (mandatory, apply to every chart)
- Set Chart.js layout: { padding: { top: 10, right: 16, bottom: 10, left: 0 } }
- Set legend: { display: false } when series count > 10
- Set maintainAspectRatio: false and let the container control height
- The chart HTML body must have: margin:0; padding:0; overflow:hidden; background:transparent
- The <canvas> must be: width:100%; height:100%; display:block
- The container div wrapping canvas must be: width:100%; height:100vh; padding:0; margin:0
- Y-axis title: set display:false — the axis values are self-explanatory
- Minimize all default Chart.js whitespace reserves
"""

    chartBlock = buildChartFrame(chartFile, vizData?.title || 'Chart');

    // ← Insert DIRECTLY into bw, before timestamp, NOT inside any bubble
    if (tsSpan && tsSpan.parentNode === bw) {
      bw.insertBefore(chartBlock, tsSpan);
    } else {
      bw.appendChild(chartBlock);
    }
    snapshot.chart = chartBlock;
  } else if (payload.error) {
    const errEl = buildErrorEl(`Chart generation failed: ${payload.error}`);
    bw.appendChild(errEl);
  }
  addOrReplaceTimestamp();
  scrollToBottom();
  break;
}

function buildChartFrame(chartFile, title) {
  const wrap = document.createElement('div');
  wrap.className = 'chart-frame-wrap';
  wrap.style.cssText = 'width:100%; background:transparent; border:none; border-radius:0; padding:0; margin-top:6px;';

  const header = document.createElement('div');
  header.className = 'chart-frame-header';
  header.innerHTML = `
    <button class="chart-open-btn"
      onclick="openChartModal('${API_BASE}/chart/${encodeURIComponent(chartFile)}', '${(title||'Chart').replace(/'/g,"\\'")}')">
      <svg viewBox="0 0 24 24" width="12" height="12" fill="currentColor">
        <path d="M19 19H5V5h7V3H5a2 2 0 00-2 2v14a2 2 0 002 2h14a2 2 0 002-2v-7h-2v7zM14 3v2h3.59l-9.83 9.83 1.41 1.41L19 6.41V10h2V3h-7z"/>
      </svg>
      Expand chart
    </button>`;

  const iframe = document.createElement('iframe');
  iframe.className = 'chart-iframe';
  iframe.src = `${API_BASE}/chart/${encodeURIComponent(chartFile)}`;
  iframe.loading = 'lazy';
  iframe.style.cssText = 'width:100%; height:420px; border:none; border-radius:var(--radius-md); display:block; background:transparent;';

  iframe.onload = () => {
    try {
      const idoc = iframe.contentDocument || iframe.contentWindow.document;
      const style = idoc.createElement('style');
      style.textContent = `
        html, body {
          margin: 0 !important; padding: 0 !important;
          overflow: hidden !important; background: transparent !important;
          width: 100% !important; height: 100% !important;
        }
        canvas { display: block !important; }
        /* Tighten Chart.js default padding */
        .chart-container, [class*="chart"] {
          padding: 0 !important;
        }
      `;
      idoc.head.appendChild(style);

      // Auto-height: use scrollHeight but cap between 320-600px
      const h = idoc.body.scrollHeight;
      if (h && h > 100) {
        iframe.style.height = Math.min(Math.max(h, 320), 600) + 'px';
      }
    } catch { /* cross-origin */ }
  };

  wrap.appendChild(header);
  wrap.appendChild(iframe);
  return wrap;
}
