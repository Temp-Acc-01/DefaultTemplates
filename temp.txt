"""
token_manager.py  --  Sarthi Token Quota System

Responsibilities:
  - Atomic token deduction from MySQL (race-condition safe)
  - Balance fetch (always live from DB, never session-cached)
  - Flask blueprint: /tokens/balance  /tokens/deduct
  - Multi-device safe: all state in DB, no in-memory caching

Table column used: api_tokens (TEXT or INT) in nlq_user_details
  - NULL or missing = unlimited (admin accounts)
  - 0 = exhausted
  - N = N tokens remaining

Integration:
  1. from token_manager import register_tokens, get_balance, deduct_tokens
  2. Call register_tokens(app) inside create_app() after register_auth(app)
  3. Frontend: GET /tokens/balance  POST /tokens/deduct
"""

WARN_THRESHOLD     = 25_000   # show warning below this
CRITICAL_THRESHOLD = 0        # block below or equal to this

# DB helpers
# ------------------------------------------------------------------

def _get_conn():
    return pymysql.connect(
        host        = MYSQL_HOST,
        port        = MYSQL_PORT,
        user        = MYSQL_USER,
        password    = MYSQL_PASS,
        database    = MYSQL_DB,
        cursorclass = pymysql.cursors.DictCursor,
        connect_timeout = 10,

def get_balance(username: str) -> dict:
    """
    Returns {available, total, unlimited, warn, blocked}
    Always reads live from DB -- safe for multi-device.

    api_tokens column stores a string like "950000/1000000"
    format: "available/total"
    OR a plain integer string "1000000" (treated as available=total on first read)
    OR NULL (unlimited)
    """
    try:
        conn = _get_conn()
        with conn.cursor() as cur:
            cur.execute(
                f"SELECT api_tokens FROM {TABLE_NAME} WHERE username = %s LIMIT 1",
                (username,)
            )
            row = cur.fetchone()
        conn.close()
    except Exception as e:
        logger.error("get_balance DB error: %s", e)
        # Fail open -- don't block user on DB error
        return {"available": 999999, "total": 999999, "unlimited": False,
                "warn": False, "blocked": False, "error": str(e)}

    if not row or row["api_tokens"] is None:
        return {"available": None, "total": None, "unlimited": True,
                "warn": False, "blocked": False}

    raw = str(row["api_tokens"]).strip()

    if "/" in raw:
        parts     = raw.split("/", 1)
        available = int(parts[0])
        total     = int(parts[1])
    else:
        # Plain integer -- treat as available = total (initialise format)
        available = int(raw)
        total     = int(raw)
        # Rewrite to standard format
        _write_balance(username, available, total)

    return {
        "available": available,
        "total":     total,
        "unlimited": False,
        "warn":      0 < available <= WARN_THRESHOLD,
        "blocked":   available <= CRITICAL_THRESHOLD,
    }


def _write_balance(username: str, available: int, total: int) -> bool:
    """Overwrite api_tokens with 'available/total' string."""
    try:
        conn = _get_conn()
        with conn.cursor() as cur:
            cur.execute(
                f"UPDATE {TABLE_NAME} SET api_tokens = %s WHERE username = %s",
                (f"{available}/{total}", username)
            )
        conn.commit()
        conn.close()
        return True
    except Exception as e:
        logger.error("_write_balance error: %s", e)
        return False


def deduct_tokens(username: str, amount: int) -> dict:
    """
    Atomically deduct `amount` tokens from user's balance.

    Uses a single UPDATE with conditional WHERE so two simultaneous
    requests from different devices cannot both succeed if balance
    would go negative.

    Returns:
      {success, available, total, warn, blocked, message}
    """
    if amount <= 0:
        bal = get_balance(username)
        return {**bal, "success": True, "message": "No deduction needed"}

    try:
        conn = _get_conn()
        with conn.cursor() as cur:
            # Read current balance first
            cur.execute(
                f"SELECT api_tokens FROM {TABLE_NAME} WHERE username = %s LIMIT 1",
                (username,)
            )
            row = cur.fetchone()

        if not row or row["api_tokens"] is None:
            conn.close()
            # Unlimited user -- no deduction needed
            return {"success": True, "available": None, "total": None,
                    "unlimited": True, "warn": False, "blocked": False,
                    "message": "Unlimited"}

        raw = str(row["api_tokens"]).strip()
        if "/" in raw:
            parts     = raw.split("/", 1)
            available = int(parts[0])
            total     = int(parts[1])
        else:
            available = int(raw)
            total     = int(raw)

        if available <= 0:
            conn.close()
            return {"success": False, "available": 0, "total": total,
                    "unlimited": False, "warn": False, "blocked": True,
                    "message": "Token quota exhausted. Please contact admin to recharge."}

        # Atomic deduction -- clamp to 0, never negative
        new_available = max(0, available - amount)
        new_raw       = f"{new_available}/{total}"

        with conn.cursor() as cur:
            affected = cur.execute(
                f"""UPDATE {TABLE_NAME}
                    SET api_tokens = %s
                    WHERE username = %s
                      AND api_tokens = %s""",
                (new_raw, username, raw)
            )
        conn.commit()
        conn.close()

        if affected == 0:
            # Concurrent update -- re-read and return current state
            logger.warning("Concurrent token deduction detected for '%s' -- re-reading", username)
            bal = get_balance(username)
            return {**bal, "success": True, "message": "Concurrent update -- balance refreshed"}

        warn    = 0 < new_available <= WARN_THRESHOLD
        blocked = new_available <= CRITICAL_THRESHOLD

        logger.info("Deducted %d tokens for '%s': %d -> %d / %d",
                    amount, username, available, new_available, total)

        return {
            "success":   True,
            "available": new_available,
            "total":     total,
            "unlimited": False,
            "warn":      warn,
            "blocked":   blocked,
            "message":   "OK",
        }

    except Exception as e:
        logger.error("deduct_tokens error for '%s': %s", username, e)
        return {"success": False, "available": None, "total": None,
                "unlimited": False, "warn": False, "blocked": False,
                "error": str(e), "message": "DB error during deduction"}

def _session_username():
    """Extract username from Flask session. Returns None if not logged in."""
    return session.get("username")


def token_login_required(f):
    @functools.wraps(f)
    def decorated(*args, **kwargs):
        if not _session_username():
            return jsonify({"status": "error", "message": "Not authenticated"}), 401
        return f(*args, **kwargs)
    return decorated


# ------------------------------------------------------------------
# Flask Blueprint
# ------------------------------------------------------------------

tokens_bp = Blueprint("tokens", __name__)


@tokens_bp.route("/tokens/balance", methods=["GET"])
@token_login_required
def balance_route():
    """
    GET /tokens/balance
    Returns current token balance for the logged-in user.
    Always reads live from DB.

    Response:
    {
      "status": "success",
      "available": 950000,
      "total": 1000000,
      "unlimited": false,
      "warn": false,
      "blocked": false
    }
    """
    username = _session_username()
    bal      = get_balance(username)
    return jsonify({"status": "success", **bal})


@tokens_bp.route("/tokens/deduct", methods=["POST"])
@token_login_required
def deduct_route():
    """
    POST /tokens/deduct
    Body: { "amount": 12345 }

    Atomically deducts tokens from the user's balance.
    Returns updated balance state.

    Response:
    {
      "status": "success",
      "available": 937655,
      "total": 1000000,
      "unlimited": false,
      "warn": false,
      "blocked": false,
      "message": "OK"
    }
    """
    username = _session_username()
    body     = request.get_json(silent=True) or {}
    amount   = int(body.get("amount") or 0)

    result = deduct_tokens(username, amount)

    status_code = 200
    if not result.get("success") and result.get("blocked"):
        status_code = 402  # Payment Required -- quota exhausted

    return jsonify({"status": "success" if result.get("success") else "error",
                    **result}), status_code

    """
    Call inside create_app() after register_auth(app).

    from token_manager import register_tokens
    register_tokens(app)
    """
    app.register_blueprint(tokens_bp)
    logger.info("Token manager registered. Warn threshold: %d, Block threshold: %d",
                WARN_THRESHOLD, CRITICAL_THRESHOLD)

def ensure_table():
    """
    Create nlq_user_details if it doesn't exist.
    Password column stores PLAIN TEXT.
    api_tokens stores quota as "available/total" string e.g. "950000/1000000"
    or NULL for unlimited users.
    """
    ddl = f"""
    CREATE TABLE IF NOT EXISTS {TABLE_NAME} (
        id          INT AUTO_INCREMENT PRIMARY KEY,
        username    VARCHAR(120) NOT NULL UNIQUE,
        password    VARCHAR(255) NOT NULL,
        api_tokens  TEXT         DEFAULT NULL,
        created_at  DATETIME     DEFAULT CURRENT_TIMESTAMP
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    """
    try:
        conn = _get_conn()
        with conn.cursor() as cur:
            cur.execute(ddl)
        conn.commit()
        conn.close()
        logger.info("Auth table '%s' ready.", TABLE_NAME)
    except Exception as e:
        logger.error("Could not ensure auth table: %s", e)


def _lookup_user(identifier: str):
    """
    Find user by username.
    Returns dict with {username, password, api_tokens} or None.
    """
    try:
        conn = _get_conn()
        with conn.cursor() as cur:
            cur.execute(
                f"SELECT username, password, api_tokens "
                f"FROM {TABLE_NAME} WHERE username = %s LIMIT 1",
                (identifier,)
            )
            row = cur.fetchone()
        conn.close()
        return row
    except Exception as e:
        logger.error("DB lookup failed: %s", e)
        return None


def _verify_password(plain: str, stored: str) -> bool:
    """
    Plain text comparison.
    Both plain and stored are compared as-is (stripped).
    """
    return plain.strip() == stored.strip()

auth_bp = Blueprint("auth", __name__)


@auth_bp.route("/auth/login", methods=["POST"])
def login():
    """
    POST /auth/login
    Body: { "identifier": "username", "password": "..." }
    Returns: { "status": "success", "username": "..." }
    """
    body       = request.get_json(silent=True) or {}
    identifier = (body.get("identifier") or "").strip()
    password   = (body.get("password")   or "").strip()

    if not identifier or not password:
        return jsonify({"status": "error",
                        "message": "Username and password are required."}), 400

    user = _lookup_user(identifier)

    if not user:
        return jsonify({"status": "error",
                        "message": "Incorrect username or password."}), 401

    if not _verify_password(password, user["password"]):
        return jsonify({"status": "error",
                        "message": "Incorrect username or password."}), 401

    session.permanent  = True
    session["username"] = user["username"]
    logger.info("User '%s' logged in.", user["username"])

    return jsonify({"status": "success", "username": user["username"]})


@auth_bp.route("/auth/logout", methods=["POST"])
def logout():
    """POST /auth/logout -- clears session."""
    username = session.pop("username", None)
    if username:
        logger.info("User '%s' logged out.", username)
    return jsonify({"status": "success"})


@auth_bp.route("/auth/me", methods=["GET"])
def me():
    """
    GET /auth/me
    Returns current user info including api_tokens for quota display.

    Response:
    {
      "status": "success",
      "username": "john",
      "api_tokens": "950000/1000000"   <- raw string, frontend parses
    }
    """
    if "username" not in session:
        return jsonify({"status": "error", "message": "Not authenticated."}), 401

    username = session["username"]

    # Always read fresh api_tokens from DB so multi-device stays in sync
    user = _lookup_user(username)
    api_tokens_raw = user["api_tokens"] if user else None

    return jsonify({
        "status":     "success",
        "username":   username,
        "api_tokens": api_tokens_raw,   # "available/total" or None
    })


# ------------------------------------------------------------------
# login_required decorator
# ------------------------------------------------------------------

def login_required(f):
    """
    Decorator for Flask route functions.
    If session has no username, returns 401 JSON for API routes
    or redirects to / for browser routes.
    """
    @functools.wraps(f)
    def decorated(*args, **kwargs):
        if "username" not in session:
            if (request.is_json or
                    request.path.startswith(("/query", "/generate", "/execute",
                                             "/tables", "/upload", "/download",
                                             "/chart", "/health", "/tokens"))):
                return jsonify({"status": "error",
                                "message": "Not authenticated."}), 401
            return redirect("/")
        return f(*args, **kwargs)
    return decorated

    """

    app.register_blueprint(auth_bp)
    ensure_table()
    logger.info("Auth module registered (plain-text passwords).")

DB column format for api_tokens:
  NULL                  = unlimited (admin)
  "1000000/1000000"     = fresh 1M quota
  "750000/1000000"      = 250K used, 750K left
  Plain int "1000000"   = auto-converted to "1000000/1000000" on first read

To add a user manually via MySQL:
  INSERT INTO nlq_user_details (username, password, api_tokens)
  VALUES ('john', 'plainpassword123', '1000000/1000000');

To top up a user:
  UPDATE nlq_user_details
  SET api_tokens = '1000000/1000000'
  WHERE username = 'john';
