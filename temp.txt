function enableTouchTableScroll() {

  const cont = document.getElementById('sdt-table-container');

  if (!cont) return;

  let isDown = false;
  let startX;
  let scrollLeft;

  cont.addEventListener('mousedown', (e) => {

    if (window.innerWidth > 900) return;

    isDown = true;

    startX = e.pageX - cont.offsetLeft;

    scrollLeft = cont.scrollLeft;
  });

  cont.addEventListener('mouseleave', () => {
    isDown = false;
  });

  cont.addEventListener('mouseup', () => {
    isDown = false;
  });

  cont.addEventListener('mousemove', (e) => {

    if (!isDown) return;

    e.preventDefault();

    const x = e.pageX - cont.offsetLeft;

    const walk = (x - startX) * 1.1;

    cont.scrollLeft = scrollLeft - walk;
  });
}
