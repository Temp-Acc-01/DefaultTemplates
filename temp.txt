function computeColWidths(columns, sampleRows) {
  // Step 1: compute natural content-based width per column
  const natural = columns.map(col => {
    let maxLen = col.name.length;
    sampleRows.slice(0, 30).forEach(r => {
      const v = String(r[col.name] || '');
      if (v.length > maxLen) maxLen = v.length;
    });
    return Math.max(COL_MIN_W, Math.min(maxLen * 7.5 + COL_PAD, 320));
  });

  // Step 2: get available container width
  const cont = document.getElementById('sdt-table-container');
  const availW = cont ? cont.clientWidth - 2 : window.innerWidth - 40;
  const totalNatural = natural.reduce((a, b) => a + b, 0);

  // Step 3: if columns don't fill the container, scale them up proportionally
  if (totalNatural < availW && columns.length > 0) {
    const scale = availW / totalNatural;
    // Cap scale so no single column gets absurdly wide (max 400px)
    return natural.map(w => Math.min(Math.floor(w * scale), 400));
  }

  return natural;
}

function openFilterPanel(colName, btnEl) {
  const col   = _state.columns.find(c => c.name === colName);
  const panel = document.getElementById('sdt-filter-panel');
  if (!col || !panel) return;

  // If same column clicked again → toggle close
  if (_state.panelOpen === colName) { closeFilterPanel(); return; }
  
  // Close any existing panel first without re-triggering listeners
  if (_state.panelOpen) {
    panel.style.display = 'none';
    _state.panelOpen = null;
    document.removeEventListener('mousedown', _outsideFilterClose);
  }

  _state.panelOpen = colName;
  _state._filterBtnEl = btnEl;  // store the button that opened it

  const rect = btnEl.getBoundingClientRect();
  panel.style.left    = Math.min(rect.left, window.innerWidth - 265) + 'px';
  panel.style.top     = (rect.bottom + 6) + 'px';
  panel.style.display = 'block';

  if (col.type === 'string') renderStringFilterPanel(col, panel);
  else renderNumericFilterPanel(col, panel);

  // Use mousedown (fires before click) so we can reliably detect outside taps
  // Small delay so the current click event doesn't immediately close the panel
  setTimeout(() => {
    document.addEventListener('mousedown', _outsideFilterClose);
  }, 50);
}

// REPLACE _outsideFilterClose:
function _outsideFilterClose(e) {
  const panel  = document.getElementById('sdt-filter-panel');
  const btnEl  = _state._filterBtnEl;
  // Close if click is outside panel AND outside the button that opened it
  if (panel && !panel.contains(e.target) && btnEl !== e.target && !btnEl?.contains(e.target)) {
    closeFilterPanel();
  }
}

// Add these two methods inside the SarthiDatatable object:

_startResize(e, ci) {
  e.preventDefault();
  e.stopPropagation();

  const startX   = e.clientX;
  const startW   = _state.colWidths[ci];
  const handle   = e.target;
  handle.classList.add('resizing');

  // Ghost line for visual feedback
  const ghost = document.createElement('div');
  ghost.style.cssText = `position:fixed;top:0;bottom:0;width:2px;background:rgba(41,121,217,0.5);
    pointer-events:none;z-index:9999;left:${e.clientX}px;`;
  document.body.appendChild(ghost);

  function onMove(ev) {
    const delta = ev.clientX - startX;
    const newW  = Math.max(80, startW + delta);
    _state.colWidths[ci] = newW;
    ghost.style.left = ev.clientX + 'px';

    // Live update all cells in this column without full re-render
    const table = document.getElementById('sdt-table');
    if (!table) return;
    table.querySelectorAll(`[data-ci="${ci}"], th[data-ci="${ci}"]`).forEach(el => {
      el.style.minWidth = newW + 'px';
      el.style.maxWidth = newW + 'px';
      el.style.width    = newW + 'px';
    });
    // Also update th directly
    const th = document.querySelector(`#sdt-thead th[data-ci="${ci}"]`);
    if (th) { th.style.minWidth=newW+'px'; th.style.maxWidth=newW+'px'; th.style.width=newW+'px'; }
  }

  function onUp() {
    handle.classList.remove('resizing');
    document.body.removeChild(ghost);
    document.removeEventListener('mousemove', onMove);
    document.removeEventListener('mouseup', onUp);
    refresh();  // full re-render to sync all cells
  }

  document.addEventListener('mousemove', onMove);
  document.addEventListener('mouseup', onUp);
},

_autoFitCol(ci) {

  const col = _state.columns[ci];

  let maxLen = col.name.length;
  _state.filteredData.slice(0, 50).forEach(r => {

  });
  _state.colWidths[ci] = Math.max(COL_MIN_W, Math.min(maxLen * 7.5 + COL_PAD, 400));
  refresh();
},
