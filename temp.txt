PATCH 6 — Update JS: updateTokenUI() and add toggleTokenBubble()
         Replace the existing updateTokenUI function in <script>

FIND:
function updateTokenUI() {
  const wrap = document.getElementById('token-balance-wrap');
  const pill = document.getElementById('token-pill');
  const txt  = document.getElementById('token-text');
  const warn = document.getElementById('token-warning');

  if (tokenState.available === null) {
    wrap.style.display = 'none';
    return;
  }

  wrap.style.display = 'flex';
  pill.className = 'token-pill' + (tokenState.warn ? ' warn' : '') + (tokenState.blocked ? ' blocked' : '');
  
  if (tokenState.unlimited || tokenState.total === null) {
    txt.textContent = '∞ tokens';
    warn.style.display = 'none';
  } else {
    txt.textContent = `${tokenState.available.toLocaleString()} / ${tokenState.total.toLocaleString()}`;
    warn.style.display = tokenState.warn ? 'block' : 'none';
    warn.textContent = tokenState.warn ? 'Low quota' : (tokenState.blocked ? 'Quota exhausted' : '');
  }
}

REPLACE WITH:
function fmtTokens(n) {
  if (n === null || n === undefined) return '—';
  if (n >= 1_000_000) return (n / 1_000_000).toFixed(2) + 'M';
  if (n >= 1_000)     return (n / 1_000).toFixed(1)     + 'K';
  return n.toLocaleString();
}

function updateTokenUI() {
  const wrap = document.getElementById('token-balance-wrap');
  const pill = document.getElementById('token-pill');
  const txt  = document.getElementById('token-text');

  if (tokenState.available === null && !tokenState.unlimited) {
    wrap.style.display = 'none';
    return;
  }

  wrap.style.display = 'flex';

  // Pill state class
  pill.className = 'token-pill'
    + (tokenState.warn    ? ' warn'    : '')
    + (tokenState.blocked ? ' blocked' : '');

  if (tokenState.unlimited || tokenState.total === null) {
    txt.textContent = '∞';
  } else {
    // Short format: e.g. "806.0K / 1.00M"
    txt.textContent = `${fmtTokens(tokenState.available)} / ${fmtTokens(tokenState.total)}`;
  }

  // Update bubble values (raw numbers for precision)
  const tibAlloc = document.getElementById('tib-allocated');
  const tibUsed  = document.getElementById('tib-used');
  const tibAvail = document.getElementById('tib-available');

  if (tibAlloc) tibAlloc.textContent = fmtTokens(tokenState.total);
  if (tibUsed)  tibUsed.textContent  = fmtTokens(
    (tokenState.total || 0) - (tokenState.available || 0)
  );
  if (tibAvail) {
    tibAvail.textContent = fmtTokens(tokenState.available);
    tibAvail.className   = 'tib-val'
      + (tokenState.blocked ? ' blocked' : tokenState.warn ? ' warn' : ' available');
  }

  // Also sync the _applyQuotaUI blocked/warn notices
  const blockedNotice = document.getElementById('quota-blocked-notice');
  const warnToast     = document.getElementById('quota-warn-toast');
  const sendBtn       = document.getElementById('send-btn');

  if (tokenState.blocked) {
    if (blockedNotice) blockedNotice.classList.add('visible');
    if (warnToast)     warnToast.classList.remove('visible');
    if (sendBtn && !isLoading) sendBtn.disabled = true;
  } else {
    if (blockedNotice) blockedNotice.classList.remove('visible');
    if (sendBtn && !isLoading) sendBtn.disabled = false;
    if (warnToast) {
      if (tokenState.warn) {
        const wt = document.getElementById('quota-warn-text');
        if (wt) wt.textContent =
          `Low quota: ${fmtTokens(tokenState.available)} remaining. Contact admin to recharge.`;
        warnToast.classList.add('visible');
      } else {
        warnToast.classList.remove('visible');
      }
    }
  }
}

// Toggle the token info bubble on ⓘ click
function toggleTokenBubble(e) {
  e.stopPropagation();
  const bubble = document.getElementById('token-info-bubble');
  if (!bubble) return;
  bubble.classList.toggle('open');
}

// Close bubble when clicking anywhere else
document.addEventListener('click', () => {
  const bubble = document.getElementById('token-info-bubble');
  if (bubble) bubble.classList.remove('open');
});

REPLACE WITH:
  async function fetchQuotaBalance() {
    try {
      const r = await fetch(`${API_BASE}/tokens/balance`);
      if (!r.ok) return;
      const d = await r.json();
      if (d.status === 'success') {
        // Normalize field names from token_manager.py response
        _applyQuotaUI({
          ...d,
          available: d.available_tokens ?? d.available,
          total:     d.total_allocated  ?? d.total,
        });
      }
    } catch (e) {
      console.warn('Token balance fetch failed:', e);
    }
  }
