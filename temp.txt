"""
auth.py
========
Login / session management module for Sarthi NL-to-SQL chat UI.

Features:
- MySQL-based user store (table: nlq_user_details)
- Password hashing with bcrypt
- Flask session-based auth
- Login / logout endpoints
- @login_required decorator for protecting routes

MySQL connection: nabh / nabh@123 / NABH
Table: nlq_user_details (username, password, api_tokens)

Integration: import and call register_auth(app) from app.py
"""

import os
import logging
import functools
import pymysql
import pymysql.cursors
import bcrypt

from flask import (
    Blueprint, request, jsonify, session,
    redirect, url_for, make_response,
)

logger = logging.getLogger(__name__)

# ─────────────────────────────────────────────────────────────────────────────
# Config
# ─────────────────────────────────────────────────────────────────────────────
MYSQL_HOST   = os.environ.get("AUTH_MYSQL_HOST",   "localhost")
MYSQL_PORT   = int(os.environ.get("AUTH_MYSQL_PORT", 3306))
MYSQL_USER   = os.environ.get("AUTH_MYSQL_USER",   "nabh")
MYSQL_PASS   = os.environ.get("AUTH_MYSQL_PASS",   "nabh@123")
MYSQL_DB     = os.environ.get("AUTH_MYSQL_DB",     "NABH")
TABLE_NAME   = "nlq_user_details"

# Flask session secret — override via env in production
SESSION_SECRET = os.environ.get("SESSION_SECRET", "sarthi-secret-change-in-prod-2026")

# ─────────────────────────────────────────────────────────────────────────────
# DB helpers
# ─────────────────────────────────────────────────────────────────────────────

def _get_conn():
    return pymysql.connect(
        host        = MYSQL_HOST,
        port        = MYSQL_PORT,
        user        = MYSQL_USER,
        password    = MYSQL_PASS,
        database    = MYSQL_DB,
        cursorclass = pymysql.cursors.DictCursor,
        connect_timeout = 10,
    )


def ensure_table():
    """
    Create nlq_user_details if it doesn't exist.
    Call once at app startup.
    Password column stores bcrypt hash (varchar 255).
    """
    ddl = f"""
    CREATE TABLE IF NOT EXISTS {TABLE_NAME} (
        id          INT AUTO_INCREMENT PRIMARY KEY,
        username    VARCHAR(120) NOT NULL UNIQUE,
        password    VARCHAR(255) NOT NULL,
        api_tokens  TEXT         DEFAULT NULL,
        created_at  DATETIME     DEFAULT CURRENT_TIMESTAMP
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    """
    try:
        conn = _get_conn()
        with conn.cursor() as cur:
            cur.execute(ddl)
        conn.commit()
        conn.close()
        logger.info("Auth table '%s' ready.", TABLE_NAME)
    except Exception as e:
        logger.error("Could not ensure auth table: %s", e)


def _lookup_user(identifier: str):
    """
    Find user by username OR email (both stored in the username column).
    Returns dict with {username, password, api_tokens} or None.
    """
    try:
        conn = _get_conn()
        with conn.cursor() as cur:
            cur.execute(
                f"SELECT username, password, api_tokens "
                f"FROM {TABLE_NAME} WHERE username = %s LIMIT 1",
                (identifier,)
            )
            row = cur.fetchone()
        conn.close()
        return row
    except Exception as e:
        logger.error("DB lookup failed: %s", e)
        return None


def _verify_password(plain: str, hashed: str) -> bool:
    """Compare plain password against bcrypt hash."""
    try:
        return bcrypt.checkpw(plain.encode(), hashed.encode())
    except Exception:
        return False


def hash_password(plain: str) -> str:
    """Utility: hash a password for inserting a new user."""
    return bcrypt.hashpw(plain.encode(), bcrypt.gensalt()).decode()


# ─────────────────────────────────────────────────────────────────────────────
# Blueprint
# ─────────────────────────────────────────────────────────────────────────────

auth_bp = Blueprint("auth", __name__)


@auth_bp.route("/auth/login", methods=["POST"])
def login():
    """
    POST /auth/login
    Body: { "identifier": "username_or_email", "password": "..." }
    Returns: { "status": "success" } or { "status": "error", "message": "..." }
    """
    body       = request.get_json(silent=True) or {}
    identifier = (body.get("identifier") or "").strip()
    password   = (body.get("password")   or "").strip()

    if not identifier or not password:
        return jsonify({"status": "error", "message": "Username and password are required."}), 400

    user = _lookup_user(identifier)

    if not user:
        return jsonify({"status": "error", "message": "Incorrect username or password."}), 401

    if not _verify_password(password, user["password"]):
        return jsonify({"status": "error", "message": "Incorrect username or password."}), 401

    # Set session
    session.permanent = True
    session["username"] = user["username"]
    logger.info("User '%s' logged in.", user["username"])

    return jsonify({"status": "success", "username": user["username"]})


@auth_bp.route("/auth/logout", methods=["POST"])
def logout():
    """POST /auth/logout — clears session."""
    username = session.pop("username", None)
    if username:
        logger.info("User '%s' logged out.", username)
    return jsonify({"status": "success"})


@auth_bp.route("/auth/me", methods=["GET"])
def me():
    """GET /auth/me — returns current user or 401."""
    if "username" in session:
        return jsonify({"status": "success", "username": session["username"]})
    return jsonify({"status": "error", "message": "Not authenticated."}), 401


# ─────────────────────────────────────────────────────────────────────────────
# login_required decorator
# ─────────────────────────────────────────────────────────────────────────────

def login_required(f):
    """
    Decorator for Flask route functions.
    If session has no username, returns 401 JSON for API routes,
    or redirects to / (login page) for browser routes.
    """
    @functools.wraps(f)
    def decorated(*args, **kwargs):
        if "username" not in session:
            # API routes (XHR) get JSON 401
            if request.is_json or request.path.startswith(("/query", "/generate", "/execute", "/tables", "/upload", "/download", "/chart", "/health")):
                return jsonify({"status": "error", "message": "Not authenticated."}), 401
            return redirect("/")
        return f(*args, **kwargs)
    return decorated


# ─────────────────────────────────────────────────────────────────────────────
# Registration helper — call from app.py
# ─────────────────────────────────────────────────────────────────────────────

def register_auth(app):
    """
    Call this from app.py's create_app() BEFORE registering other routes.

    Sets secret key, registers auth blueprint, ensures DB table exists,
    and wraps all data routes with login_required.
    """
    app.secret_key = SESSION_SECRET

    # Register blueprint
    app.register_blueprint(auth_bp)

    # Ensure table exists
    ensure_table()

    logger.info("Auth module registered.")

<script>
  const API_BASE = window.location.origin;

  // ── Check session on load ─────────────────────────────────────────────────
  // If already logged in, redirect to chat immediately
  (async () => {
    try {
      const r = await fetch(`${API_BASE}/auth/me`);
      if (r.ok) {
        const d = await r.json();
        if (d.status === 'success') {
          window.location.href = '/chat';
        }
      }
    } catch { /* not logged in */ }
  })();

  // ── Password visibility toggle ────────────────────────────────────────────
  function togglePw() {
    const inp    = document.getElementById('password');
    const open   = document.getElementById('pw-eye-open');
    const closed = document.getElementById('pw-eye-closed');
    const show   = inp.type === 'password';
    inp.type     = show ? 'text' : 'password';
    open.style.display   = show ? 'none'  : '';
    closed.style.display = show ? ''      : 'none';
  }

  // ── Error display ─────────────────────────────────────────────────────────
  function showError(msg) {
    const el   = document.getElementById('error-msg');
    const text = document.getElementById('error-text');
    text.textContent = msg;
    el.classList.remove('show');
    // Force reflow to retrigger shake animation
    void el.offsetWidth;
    el.classList.add('show');
  }

  function clearError() {
    document.getElementById('error-msg').classList.remove('show');
  }

  // ── Login handler ─────────────────────────────────────────────────────────
  async function handleLogin(e) {
    e.preventDefault();
    clearError();

    const identifier = document.getElementById('identifier').value.trim();
    const password   = document.getElementById('password').value.trim();
    const btn        = document.getElementById('login-btn');

    if (!identifier) { showError('Please enter your username or email.'); return; }
    if (!password)   { showError('Please enter your password.'); return; }

    // Loading state
    btn.classList.add('loading');
    btn.disabled = true;

    try {
      const resp = await fetch(`${API_BASE}/auth/login`, {
        method:  'POST',
        headers: { 'Content-Type': 'application/json' },
        body:    JSON.stringify({ identifier, password }),
      });

      const data = await resp.json();

      if (resp.ok && data.status === 'success') {
        // Redirect to chat
        window.location.href = '/chat';
      } else {
        showError(data.message || 'Incorrect username or password.');
      }
    } catch {
      showError('Cannot connect to server. Please try again.');
    } finally {
      btn.classList.remove('loading');
      btn.disabled = false;
    }
  }

  // Clear error when user starts typing
  document.getElementById('identifier').addEventListener('input', clearError);
  document.getElementById('password').addEventListener('input', clearError);
</script>

# ── STEP 6: sarthi_index.html — JavaScript to ADD inside <script> ────────

LOGOUT_JS = """
// ─── Auth / logout ───────────────────────────────────────────────────────
let _currentUsername = '';

// Fetch current user on load and store username for display
(async () => {
  try {
    const r = await fetch(`${API_BASE}/auth/me`);
    if (!r.ok) {
      // Not authenticated — redirect to login
      window.location.href = '/';
      return;
    }
    const d = await r.json();
    if (d.status === 'success') {
      _currentUsername = d.username;
    }
  } catch {
    window.location.href = '/';
  }
})();

function confirmLogout() {
  document.getElementById('logout-username-display').textContent =
    _currentUsername || 'you';
  document.getElementById('logout-confirm-overlay').classList.add('open');
  document.body.style.overflow = 'hidden';
}

function closeLogoutConfirm() {
  document.getElementById('logout-confirm-overlay').classList.remove('open');
  document.body.style.overflow = '';
}

async function doLogout() {
  try {
    await fetch(`${API_BASE}/auth/logout`, { method: 'POST' });
  } catch { /* ignore */ }
  closeLogoutConfirm();
  window.location.href = '/';
}

// Close logout modal on backdrop click
document.getElementById('logout-confirm-overlay')?.addEventListener('click', function(e) {
  if (e.target === this) closeLogoutConfirm();
});

// Also close on Escape (add to existing keydown handler or add new one)
document.addEventListener('keydown', e => {
  if (e.key === 'Escape') {
    closeLogoutConfirm();
    closeChartModal(null);   // already exists
  }
});
"""

CREATE_USER_SCRIPT = """
# create_user.py — run once to add users
import pymysql
import bcrypt

conn = pymysql.connect(
    host='localhost', port=3306,
    user='nabh', password='nabh@123',
    database='NABH',
    cursorclass=pymysql.cursors.DictCursor
)

def add_user(username, plain_password):
    hashed = bcrypt.hashpw(plain_password.encode(), bcrypt.gensalt()).decode()
    with conn.cursor() as cur:
        cur.execute(
            \"INSERT INTO nlq_user_details (username, password) VALUES (%s, %s)\",
            (username, hashed)
        )
    conn.commit()
    print(f\"User '{username}' created.\")

add_user('admin', 'your_password_here')
add_user('gaurav', 'gaurav_password_here')
conn.close()
"""
