@app.route("/query/stream", methods=["POST", "GET"])
@login_required    
def query_stream():
    """
    Streaming version of /query using Server-Sent Events.
    NEW: Integrates token reservation/refund pattern.
    """
    from flask import Response, stream_with_context
    import json

    # ── Extract request params ──
    if request.method == "GET":
        question    = (request.args.get("question") or "").strip()
        max_retries = int(request.args.get("max_retries") or MAX_RETRIES)
        pinned      = PINNED_TABLES
    else:
        body        = request.get_json(silent=True) or {}
        question    = (body.get("question") or "").strip()
        max_retries = int(body.get("max_retries") or MAX_RETRIES)
        pinned      = PINNED_TABLES

    # ── Validate question ──
    if not question:
        def _err_gen():
            yield f"event: pipeline_error\ndata: {json.dumps({'error': 'question is required'})}\n\n"
            yield "event: done\ndata: {}\n\n"
        return Response(stream_with_context(_err_gen()),
                        mimetype="text/event-stream",
                        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})

    if not pipeline.has_tables():
        def _err_gen():
            yield f"event: pipeline_error\ndata: {json.dumps({'error': 'No tables loaded'})}\n\n"
            yield "event: done\ndata: {}\n\n"
        return Response(stream_with_context(_err_gen()),
                        mimetype="text/event-stream",
                        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})

    # ── Token Management: Get user info ──
    username = session.get("username")
    request_id = str(uuid.uuid4())  # Unique ID for this request

    def generate():
        try:
            yield ": keep-alive\n\n"   # first byte flushes headers immediately
            
            # ── Call pipeline with token tracking ──
            for chunk in pipeline.stream_query(
                question=question,
                pinned_tables=pinned,
                max_retries=max_retries,
                username=username,        # ← NEW: for token tracking
                request_id=request_id,    # ← NEW: for transaction logging
            ):
                yield chunk
                
        except Exception as exc:
            import traceback
            logger.error("stream_query error: %s", traceback.format_exc())
            yield f"event: pipeline_error\ndata: {json.dumps({'error': str(exc)})}\n\n"
            yield "event: done\ndata: {}\n\n"

    return Response(
        stream_with_context(generate()),
        mimetype="text/event-stream",
        headers={
            "Cache-Control":    "no-cache",
            "X-Accel-Buffering": "no",   # disables nginx buffering
            "Connection":       "keep-alive",
        },
    )
