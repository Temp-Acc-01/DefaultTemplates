# create_user.py - FIXED VERSION
import pymysql
import os
from dotenv import load_dotenv
from token_manager import allocate_tokens, _get_conn, get_wallet, ensure_wallet

load_dotenv()

USERNAME = "user2"
PASSWORD = "user123"
TOKENS   = 1_000_000

print(f"🔍 Starting user creation for '{USERNAME}'...")

conn = _get_conn()
try:
    with conn.cursor() as cur:
        # 1. Check if user exists
        cur.execute("SELECT id FROM nlq_user_details WHERE username = %s", (USERNAME,))
        user_row = cur.fetchone()
        
        if user_row:
            user_id = user_row['id']
            print(f"⚠️  User '{USERNAME}' already exists (ID: {user_id})")
        else:
            # Create user
            print(f"➕ Creating user '{USERNAME}'...")
            cur.execute(
                "INSERT INTO nlq_user_details (username, password) VALUES (%s, %s)",
                (USERNAME, PASSWORD)
            )
            user_id = cur.lastrowid
            conn.commit()
            print(f"✅ User created with ID: {user_id}")

    # 2. Allocate tokens (creates wallet if needed)
    print(f"💰 Allocating {TOKENS:,} tokens...")
    wallet = allocate_tokens(user_id, TOKENS, request_id="manual_setup")
    
    # ✅ Check if allocation worked by looking at available_tokens
    if wallet and wallet.get('available_tokens', 0) >= TOKENS:
        print(f"✅ SUCCESS! Tokens allocated to '{USERNAME}'")
        print(f"   Balance: {wallet['available_tokens']:,}")
        print(f"   Total allocated: {wallet['total_allocated']:,}")
    else:
        print(f"⚠️  Allocation may have failed. Current balance: {wallet}")

except Exception as e:
    import traceback
    print(f"❌ Failed: {e}")
    traceback.print_exc()
    conn.rollback()
finally:
    conn.close()
