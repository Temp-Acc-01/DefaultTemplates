Database Tables:

- nlq_token_transactions: audit ledger

def _get_conn():
    """Get database connection with DictCursor."""
    return pymysql.connect(
        host=MYSQL_HOST,
        port=MYSQL_PORT,
        user=MYSQL_USER,
        password=MYSQL_PASS,
        database=MYSQL_DB,
        cursorclass=pymysql.cursors.DictCursor,
        autocommit=False,  # Manual transaction control
        connect_timeout=10,
    )


@contextmanager
def _transaction():
    """Context manager for database transactions."""
    conn = _get_conn()
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def _get_user_id(username: str) -> Optional[int]:
    """Get user ID from username."""
    with _transaction() as conn:
        with conn.cursor() as cur:
            cur.execute(
                "SELECT id FROM nlq_user_details WHERE username = %s LIMIT 1",
                (username,)
            )
            row = cur.fetchone()
            return row['id'] if row else None

def get_wallet(user_id: int) -> Dict[str, Any]:
    """
    Get wallet balance for user.
    Returns wallet data or None if user has no wallet.
    """
    with _transaction() as conn:
        with conn.cursor() as cur:
            cur.execute("""
                SELECT user_id, total_allocated, total_used, 
                       available_tokens, updated_at
                FROM nlq_token_wallets
                WHERE user_id = %s
            """, (user_id,))
            row = cur.fetchone()
            
            if not row:
                return None
            
            available = row['available_tokens']
            return {
                'user_id': row['user_id'],
                'total_allocated': row['total_allocated'],
                'total_used': row['total_used'],
                'available_tokens': available,
                'updated_at': row['updated_at'],
                'unlimited': False,
                'warn': 0 < available <= WARN_THRESHOLD,
                'blocked': available <= CRITICAL_THRESHOLD,
            }


def ensure_wallet(user_id: int, initial_tokens: int = 0) -> Dict[str, Any]:
    """
    Ensure user has a wallet. Create if doesn't exist.
    """
    with _transaction() as conn:
        with conn.cursor() as cur:
            # Try to insert, ignore if exists
            cur.execute("""
                INSERT IGNORE INTO nlq_token_wallets
                (user_id, total_allocated, total_used, available_tokens, updated_at)
                VALUES (%s, %s, 0, %s, NOW())
            """, (user_id, initial_tokens, initial_tokens))
            
            # If inserted, log the allocation
            if cur.rowcount > 0:
                cur.execute("""
                    INSERT INTO nlq_token_transactions
                    (user_id, type, tokens, balance_before, balance_after, metadata)
                    VALUES (%s, 'ALLOCATE', %s, 0, %s, 
                            JSON_OBJECT('note', 'Initial wallet creation'))
                """, (user_id, initial_tokens, initial_tokens))
            
            conn.commit()
    
    return get_wallet(user_id)


def allocate_tokens(user_id: int, amount: int, request_id: str = None) -> Dict[str, Any]:
    """
    Allocate tokens to user (admin operation).
    Uses atomic update with transaction logging.
    """
    if amount <= 0:
        raise ValueError("Allocation amount must be positive")
    
    with _transaction() as conn:
        with conn.cursor() as cur:
            # Get current balance
            cur.execute("""
                SELECT available_tokens, total_allocated
                FROM nlq_token_wallets
                WHERE user_id = %s FOR UPDATE
            """, (user_id,))
            row = cur.fetchone()
            
            if not row:
                # Create wallet if doesn't exist
                cur.execute("""
                    INSERT INTO nlq_token_wallets
                    (user_id, total_allocated, available_tokens, updated_at)
                    VALUES (%s, %s, %s, NOW())
                """, (user_id, amount, amount))
                balance_before = 0
                balance_after = amount
            else:
                balance_before = row['available_tokens']
                new_allocated = row['total_allocated'] + amount
                new_available = row['available_tokens'] + amount
                
                # Atomic update
                cur.execute("""
                    UPDATE nlq_token_wallets
                    SET total_allocated = %s,
                        available_tokens = %s,
                        updated_at = NOW()
                    WHERE user_id = %s
                """, (new_allocated, new_available, user_id))
                
                balance_after = new_available
            
            # Log transaction
            cur.execute("""
                INSERT INTO nlq_token_transactions
                (user_id, type, tokens, request_id, balance_before, balance_after, metadata)
                VALUES (%s, 'ALLOCATE', %s, %s, %s, %s, 
                        JSON_OBJECT('admin_operation', true))
            """, (user_id, amount, request_id, balance_before, balance_after))
            
            conn.commit()
    
    logger.info("Allocated %d tokens to user %d: %d → %d", 
                amount, user_id, balance_before, balance_after)
    
    return get_wallet(user_id)


def deduct_tokens(user_id: int, amount: int, request_id: str = None,
                  model_name: str = None, metadata: Dict = None) -> Dict[str, Any]:
    """
    Deduct tokens atomically.
    
    CRITICAL: Uses conditional UPDATE to prevent race conditions.
    Only succeeds if user has sufficient balance.
    
    Returns:
        {success: bool, wallet: dict, message: str}
    """
    if amount <= 0:
        wallet = get_wallet(user_id)
        return {'success': True, 'wallet': wallet, 'message': 'No deduction needed'}
    
    with _transaction() as conn:
        with conn.cursor() as cur:
            # Get current balance
            cur.execute("""
                SELECT available_tokens, total_allocated, total_used
                FROM nlq_token_wallets
                WHERE user_id = %s FOR UPDATE
            """, (user_id,))
            row = cur.fetchone()
            
            if not row:
                return {
                    'success': False,
                    'wallet': None,
                    'message': 'User has no wallet'
                }
            
            balance_before = row['available_tokens']
            
            if balance_before < amount:
                return {
                    'success': False,
                    'wallet': get_wallet(user_id),
                    'message': f'Insufficient balance: {balance_before} < {amount}'
                }
            
            # Atomic deduction
            new_available = balance_before - amount
            new_used = row['total_used'] + amount
            
            cur.execute("""
                UPDATE nlq_token_wallets
                SET available_tokens = %s,
                    total_used = %s,
                    updated_at = NOW()
                WHERE user_id = %s AND available_tokens >= %s
            """, (new_available, new_used, user_id, amount))
            
            if cur.rowcount == 0:
                # Race condition: another request deducted first
                conn.rollback()
                logger.warning("Concurrent deduction detected for user %d", user_id)
                return {
                    'success': False,
                    'wallet': get_wallet(user_id),
                    'message': 'Concurrent update -- please retry'
                }
            
            # Log transaction
            meta_json = json.dumps(metadata) if metadata else '{}'
            cur.execute("""
                INSERT INTO nlq_token_transactions
                (user_id, type, tokens, request_id, model_name, 
                 balance_before, balance_after, metadata)
                VALUES (%s, 'DEDUCT', %s, %s, %s, %s, %s, %s)
            """, (user_id, amount, request_id, model_name, 
                  balance_before, new_available, meta_json))
            
            conn.commit()
    
    logger.info("Deducted %d tokens from user %d: %d → %d",
                amount, user_id, balance_before, new_available)
    
    return {
        'success': True,
        'wallet': get_wallet(user_id),
        'message': 'OK'
    }


def reserve_tokens(user_id: int, amount: int, request_id: str,
                   model_name: str = None) -> Dict[str, Any]:
    """
    Reserve tokens for an operation (like streaming query).
    Tokens are deducted upfront and can be refunded later.
    
    This prevents abuse and ensures user has sufficient balance
    before starting expensive operations.
    """
    result = deduct_tokens(
        user_id=user_id,
        amount=amount,
        request_id=request_id,
        model_name=model_name,
        metadata={'operation': 'reserve'}
    )
    
    if result['success']:
        logger.info("Reserved %d tokens for request %s", amount, request_id)
    
    return result


def refund_tokens(user_id: int, amount: int, request_id: str,
                  reason: str = 'unused') -> Dict[str, Any]:
    """
    Refund reserved tokens (partial or full).
    Used when actual usage is less than reserved amount.
    """
    if amount <= 0:
        return {'success': True, 'wallet': get_wallet(user_id), 'message': 'Nothing to refund'}
    
    with _transaction() as conn:
        with conn.cursor() as cur:
            # Get current balance
            cur.execute("""
                SELECT available_tokens
                FROM nlq_token_wallets
                WHERE user_id = %s FOR UPDATE
            """, (user_id,))
            row = cur.fetchone()
            
            if not row:
                return {'success': False, 'wallet': None, 'message': 'User has no wallet'}
            
            balance_before = row['available_tokens']
            new_available = balance_before + amount
            
            cur.execute("""
                UPDATE nlq_token_wallets
                SET available_tokens = %s,
                    updated_at = NOW()
                WHERE user_id = %s
            """, (new_available, user_id))
            
            # Log refund transaction
            cur.execute("""
                INSERT INTO nlq_token_transactions
                (user_id, type, tokens, request_id, balance_before, balance_after, metadata)
                VALUES (%s, 'REFUND', %s, %s, %s, %s, JSON_OBJECT('reason', %s))
            """, (user_id, amount, request_id, balance_before, new_available, reason))
            
            conn.commit()
    
    logger.info("Refunded %d tokens to user %d (request %s): %d → %d",
                amount, user_id, request_id, balance_before, new_available)
    
    return {
        'success': True,
        'wallet': get_wallet(user_id),
        'message': f'Refunded {amount} tokens'
    }


def adjust_tokens(user_id: int, reserved: int, actual: int, 
                  request_id: str, model_name: str = None) -> Dict[str, Any]:
    """
    Adjust tokens after operation completes.
    If actual < reserved: refund the difference
    If actual > reserved: deduct the difference
    
    This is the final step in the reserve/refund pattern.
    """
    if actual == reserved:
        # Perfect match - just update the transaction log
        return {
            'success': True,
            'wallet': get_wallet(user_id),
            'message': 'No adjustment needed',
            'adjusted': 0
        }
    
    difference = reserved - actual
    
    if difference > 0:
        # Refund unused tokens
        result = refund_tokens(
            user_id=user_id,
            amount=difference,
            request_id=request_id,
            reason='unused_tokens'
        )
        result['adjusted'] = difference
        result['message'] = f'Refunded {difference} unused tokens'
    else:
        # Deduct additional tokens (shouldn't happen often with good estimation)
        result = deduct_tokens(
            user_id=user_id,
            amount=abs(difference),
            request_id=request_id,
            model_name=model_name,
            metadata={'operation': 'adjustment', 'reason': 'underestimated'}
        )
        result['adjusted'] = difference
        result['message'] = f'Deducted {abs(difference)} additional tokens'
    
    return result


# ─────────────────────────────────────────────────────────────────
# Token Estimation (for Reserve Pattern)
# ─────────────────────────────────────────────────────────────────

def estimate_query_tokens(question: str) -> int:
    """
    Estimate tokens needed for a query.
    Simple heuristic: ~4 chars per token + overhead.
    
    For production, you might want to:
    - Use tiktoken library for accurate estimation
    - Track historical usage per user/query type
    - Adjust based on complexity
    """
    # Rough estimate: question length + SQL generation overhead
    base_estimate = len(question) // 3  # ~3 chars per token average
    
    # Add overhead for SQL generation, validation, visualization
    overhead = 5000  # Base overhead
    
    # Add buffer (50% extra)
    estimate = int((base_estimate + overhead) * 1.5)
    
    # Cap between min and max
    min_tokens = 10000
    max_tokens = 100000
    
    return max(min_tokens, min(estimate, max_tokens))

tokens_bp = Blueprint("tokens", __name__)


def _session_username() -> Optional[str]:
    """Get username from Flask session."""
    return session.get("username")


def _session_user_id() -> Optional[int]:
    """Get user ID from username in session."""
    username = _session_username()
    if not username:
        return None
    return _get_user_id(username)


def token_login_required(f):
    """Decorator to require authentication."""
    @functools.wraps(f)
    def decorated(*args, **kwargs):
        if not _session_username():
            return jsonify({
                "status": "error",
                "message": "Not authenticated"
            }), 401
        return f(*args, **kwargs)
    return decorated


@tokens_bp.route("/tokens/balance", methods=["GET"])
@token_login_required
def balance_route():
    """
    GET /tokens/balance
    Returns current token balance for logged-in user.
    """
    user_id = _session_user_id()
    if not user_id:
        return jsonify({
            "status": "error",
            "message": "User not found"
        }), 404
    
    wallet = get_wallet(user_id)
    if not wallet:
        wallet = ensure_wallet(user_id, initial_tokens=0)
    
    return jsonify({
        "status": "success",
        **wallet
    })

@tokens_bp.route("/tokens/allocate", methods=["POST"])
@token_login_required
def allocate_route():
    """
    POST /tokens/allocate
    Body: { "username": "user123", "amount": 100000 }
    
    Admin-only endpoint to allocate tokens to a user.
    In production, add admin role check here.
    """
    body = request.get_json(silent=True) or {}
    username = body.get("username")
    amount = int(body.get("amount") or 0)
    request_id = body.get("request_id") or str(uuid.uuid4())
    
    if not username:
        return jsonify({
            "status": "error",
            "message": "username required"
        }), 400
    
    user_id = _get_user_id(username)
    if not user_id:
        return jsonify({
            "status": "error",
            "message": f"User '{username}' not found"
        }), 404
    
    try:
        result = allocate_tokens(
            user_id=user_id,
            amount=amount,
            request_id=request_id
        )
        return jsonify({
            "status": "success",
            **result
        })
    except Exception as e:
        logger.error("Allocation failed: %s", e)
        return jsonify({
            "status": "error",
            "message": str(e)
        }), 500


@tokens_bp.route("/tokens/transactions", methods=["GET"])
@token_login_required
def transactions_route():
    """
    GET /tokens/transactions?limit=50&offset=0
    Returns transaction history for current user.
    """
    user_id = _session_user_id()
    limit = min(int(request.args.get("limit") or 50), 100)
    offset = int(request.args.get("offset") or 0)
    
    with _transaction() as conn:
        with conn.cursor() as cur:
            cur.execute("""
                SELECT id, type, tokens, request_id, model_name,
                       balance_before, balance_after, metadata, created_at
                FROM nlq_token_transactions
                WHERE user_id = %s
                ORDER BY created_at DESC
                LIMIT %s OFFSET %s
            """, (user_id, limit, offset))
            
            transactions = cur.fetchall()
            
            # Get total count
            cur.execute("""
                SELECT COUNT(*) as total
                FROM nlq_token_transactions
                WHERE user_id = %s
            """, (user_id,))
            total = cur.fetchone()['total']
    
    # Convert datetime to ISO format
    for tx in transactions:
        if tx['created_at']:
            tx['created_at'] = tx['created_at'].isoformat()
    
    return jsonify({
        "status": "success",
        "transactions": transactions,
        "total": total,
        "limit": limit,
        "offset": offset
    })
