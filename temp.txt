// ─── Sidebar open/close (tablet/mobile) ──────────────────────────
function openSidebar() {
  document.getElementById('sidebar').classList.add('open');
  document.getElementById('sidebar-overlay').classList.add('visible');
  document.body.style.overflow = 'hidden';
}

function closeSidebar() {
  const sidebar = document.getElementById('sidebar');
  sidebar.classList.remove('open');
  sidebar.classList.remove('collapsed');
  document.getElementById('sidebar-overlay').classList.remove('visible');
  document.body.style.overflow = '';
}

// Detect viewport and auto-collapse sidebar on small screens
function handleViewportResize() {
  const sidebar = document.getElementById('sidebar');
  const toggleBtn = document.getElementById('sidebar-toggle-btn');
  const prefix = document.getElementById('topbar-session-prefix');
  const w = window.innerWidth;

  if (w <= 1024) {
    // Tablet/mobile: sidebar is overlay, toggle button visible
    if (toggleBtn) toggleBtn.style.display = 'flex';
    // On mobile default closed, on tablet default closed too
    if (!sidebar.classList.contains('open')) {
      sidebar.classList.add('collapsed');
    }
  } else {
    // Desktop: sidebar always visible
    sidebar.classList.remove('collapsed', 'open');
    document.getElementById('sidebar-overlay').classList.remove('visible');
    if (toggleBtn) toggleBtn.style.display = 'none';
    document.body.style.overflow = '';
  }
}

// Keep session prefix in topbar in sync with session title
function syncTopbarPrefix() {
  const title = document.getElementById('session-title');
  const prefix = document.getElementById('topbar-session-prefix');
  if (prefix && title) prefix.textContent = title.textContent;
}

window.addEventListener('resize', handleViewportResize);
handleViewportResize();

// Patch newChat and switchSession to sync prefix
const _origNewChat = newChat;
newChat = function() {
  _origNewChat();
  syncTopbarPrefix();
};
const _origSwitchSession = switchSession;
switchSession = function(id) {
  _origSwitchSession(id);
  syncTopbarPrefix();
  // Close sidebar on mobile after selecting session
  if (window.innerWidth <= 1024) closeSidebar();
};
