/**
 * datatable_module.js  —  Sarthi Rich Datatable  v2
 *
 * Fixes in v2:
 *  1. Reduced top padding / compact stats strip
 *  2. Stats cards width-aligned to column headers (shared scroll)
 *  3. Filter icon → 3-line slider icon (interpretable)
 *  4. Histogram: min/max labels at bottom corners, dotted mean(blue) +
 *     median(purple) vertical lines with hover tooltip
 *  5. Sort icon: 3-state always-visible (neutral ↕ faded | asc ↑ blue | desc ↓ blue)
 *     cycling: none → asc → desc → none
 *
 * Usage:
 *   SarthiDatatable.open(csvUrl, title)
 *   SarthiDatatable.close()
 */

(function (global) {
  'use strict';

  const VIRTUAL_ROW_H  = 34;
  const VIRTUAL_BUFFER = 40;
  const COL_MIN_W      = 140;
  const CHARTJS_CDN    = 'https://cdn.jsdelivr.net/npm/chart.js@4.4.0/dist/chart.umd.min.js';

  /* ── STATE ── */
  let _st = {
    rawData: [], columns: [], filteredData: [], filters: {},
    sortCol: null, sortDir: 'none',
    csvUrl: null, title: '',
    sparkCharts: {}, panelOpen: null, globalSearch: '', colWidths: [],
  };

  /* ── CSV PARSER ── */
  function parseCSV(text) {
    const lines = text.split(/\r?\n/);
    if (lines.length < 2) return { headers: [], rows: [] };
    const headers = splitLine(lines[0]);
    const rows = [];
    for (let i = 1; i < lines.length; i++) {
      if (!lines[i].trim()) continue;
      const vals = splitLine(lines[i]);
      const obj  = {};
      headers.forEach((h, j) => { obj[h] = vals[j] ?? ''; });
      rows.push(obj);
    }
    return { headers, rows };
  }
  function splitLine(line) {
    const r = []; let cur = '', inQ = false;
    for (const ch of line) {
      if (ch === '"') { inQ = !inQ; continue; }
      if (ch === ',' && !inQ) { r.push(cur.trim()); cur = ''; continue; }
      cur += ch;
    }
    r.push(cur.trim()); return r;
  }

  /* ── TYPE DETECTION ── */
  function detectColumns(headers, rows) {
    return headers.map(name => {
      const vals    = rows.map(r => r[name]).filter(v => v !== '' && v != null);
      const numVals = vals.map(v => parseFloat(v)).filter(v => !isNaN(v));
      const isNum   = numVals.length / Math.max(vals.length, 1) > 0.85;
      if (isNum) {
        numVals.sort((a, b) => a - b);
        const min = numVals[0] ?? 0, max = numVals[numVals.length - 1] ?? 0;
        const mean = numVals.reduce((a, b) => a + b, 0) / (numVals.length || 1);
        const mid  = Math.floor(numVals.length / 2);
        const median = numVals.length % 2 === 0
          ? (numVals[mid - 1] + numVals[mid]) / 2 : numVals[mid];
        return { name, type: 'numeric', uniqueVals: [],
          stats: { min, max, mean, median, nullCount: rows.length - vals.length, hist: buildHist(numVals, 12) } };
      }
      const freq = {};
      vals.forEach(v => { freq[v] = (freq[v] || 0) + 1; });
      const uniqueVals = Object.entries(freq).sort((a, b) => b[1] - a[1]).map(([val, count]) => ({ val, count }));
      return { name, type: 'string', uniqueVals,
        stats: { uniqueCount: uniqueVals.length, nullCount: rows.length - vals.length } };
    });
  }
  function buildHist(sorted, bins) {
    if (!sorted.length) return { labels: [], counts: [] };
    const mn = sorted[0], mx = sorted[sorted.length - 1];
    const step = (mx - mn) / bins || 1;
    const counts = new Array(bins).fill(0);
    const labels = Array.from({ length: bins }, (_, i) => +(mn + i * step).toFixed(2));
    sorted.forEach(v => { counts[Math.min(Math.floor((v - mn) / step), bins - 1)]++; });
    return { labels, counts };
  }

  /* ── FILTER ENGINE ── */
  function applyFilters() {
    let data = _st.rawData;
    if (_st.globalSearch) {
      const q = _st.globalSearch.toLowerCase();
      data = data.filter(r => Object.values(r).some(v => String(v).toLowerCase().includes(q)));
    }
    _st.columns.forEach(col => {
      const f = _st.filters[col.name]; if (!f) return;
      if (col.type === 'string') {
        if (f.selected && f.selected.length) data = data.filter(r => f.selected.includes(r[col.name]));
        if (f.text) { const q = f.text.toLowerCase(); data = data.filter(r => String(r[col.name]).toLowerCase().includes(q)); }
      } else {
        if (f.min !== '' && f.min != null) data = data.filter(r => parseFloat(r[col.name]) >= parseFloat(f.min));
        if (f.max !== '' && f.max != null) data = data.filter(r => parseFloat(r[col.name]) <= parseFloat(f.max));
      }
    });
    if (_st.sortCol && _st.sortDir !== 'none') {
      const col = _st.columns.find(c => c.name === _st.sortCol);
      const dir = _st.sortDir === 'asc' ? 1 : -1;
      data = [...data].sort((a, b) => {
        const av = a[_st.sortCol], bv = b[_st.sortCol];
        return col && col.type === 'numeric'
          ? (parseFloat(av) - parseFloat(bv)) * dir
          : String(av).localeCompare(String(bv)) * dir;
      });
    }
    _st.filteredData = data;
  }

  /* ── COLUMN WIDTHS ── */
  function computeColWidths() {
    _st.colWidths = _st.columns.map(col =>
      Math.max(COL_MIN_W, Math.min(col.name.length * 8 + 90, 260))
    );
  }

  /* ── VIRTUAL ROWS ── */
  function renderVirtualRows(tbody, cont) {
    const data = _st.filteredData, cols = _st.columns;
    const sTop = cont.scrollTop, viewH = cont.clientHeight;
    const s    = Math.max(0, Math.floor(sTop / VIRTUAL_ROW_H) - VIRTUAL_BUFFER);
    const e    = Math.min(data.length, s + Math.ceil(viewH / VIRTUAL_ROW_H) + VIRTUAL_BUFFER * 2);
    const topH = s * VIRTUAL_ROW_H, botH = (data.length - e) * VIRTUAL_ROW_H;
    let html   = `<tr style="height:${topH}px"><td colspan="${cols.length}" style="padding:0;border:none"></td></tr>`;
    for (let i = s; i < e; i++) {
      const row = data[i];
      html += `<tr class="dt-row ${i % 2 === 0 ? 'dt-re' : 'dt-ro'}">`;
      cols.forEach((col, ci) => {
        const w = _st.colWidths[ci];
        const v = row[col.name] ?? '';
        html += `<td class="dt-cell" style="width:${w}px;min-width:${w}px;max-width:${w}px" title="${esc(String(v))}">${v === '' ? '<span class="dt-null">—</span>' : esc(String(v))}</td>`;
      });
      html += '</tr>';
    }
    html += `<tr style="height:${botH}px"><td colspan="${cols.length}" style="padding:0;border:none"></td></tr>`;
    tbody.innerHTML = html;
  }

  /* ── SVG ICONS ── */
  function iconFilter(active) {
    const c = active ? '#2979D9' : '#7A8BA8';
    return `<svg width="13" height="13" viewBox="0 0 20 20" fill="none" stroke="${c}" stroke-width="1.8" stroke-linecap="round">
      <line x1="2" y1="5"  x2="18" y2="5"/><circle cx="7"  cy="5"  r="2" fill="${c}" stroke="none"/>
      <line x1="2" y1="10" x2="18" y2="10"/><circle cx="13" cy="10" r="2" fill="${c}" stroke="none"/>
      <line x1="2" y1="15" x2="18" y2="15"/><circle cx="9"  cy="15" r="2" fill="${c}" stroke="none"/>
    </svg>`;
  }

  function iconSort(colName) {
    const active = _st.sortCol === colName;
    const dir    = active ? _st.sortDir : 'none';
    if (dir === 'asc')  return `<svg width="13" height="13" viewBox="0 0 20 20" fill="none" stroke="#2979D9" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" title="Ascending"><path d="M10 4l-4 5h8l-4-5z" fill="#2979D9" stroke="none"/><line x1="5" y1="15" x2="15" y2="15" stroke="#2979D9"/></svg>`;
    if (dir === 'desc') return `<svg width="13" height="13" viewBox="0 0 20 20" fill="none" stroke="#2979D9" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" title="Descending"><path d="M10 16l4-5H6l4 5z" fill="#2979D9" stroke="none"/><line x1="5" y1="5" x2="15" y2="5" stroke="#2979D9"/></svg>`;
    return `<svg width="13" height="13" viewBox="0 0 20 20" fill="none" title="Sort"><path d="M10 4l-3 4h6z" fill="rgba(100,130,180,0.4)"/><path d="M10 16l3-4H7z" fill="rgba(100,130,180,0.4)"/></svg>`;
  }

  /* ── HEADER ── */
  function renderHeader() {
    const thead = document.getElementById('sdt-thead');
    if (!thead) return;
    let html = '<tr>';
    _st.columns.forEach((col, ci) => {
      const w   = _st.colWidths[ci];
      const cls = col.type === 'numeric' ? 'numeric' : 'string';
      html += `<th class="dt-th" style="width:${w}px;min-width:${w}px;max-width:${w}px">
        <div class="dt-thi">
          <span class="dt-tb ${cls}">${col.type === 'numeric' ? 'NUM' : 'STR'}</span>
          <span class="dt-thn" title="${esc(col.name)}">${esc(col.name)}</span>
          <button class="dt-sort-btn" title="Sort" onclick="SarthiDatatable._sortBy('${esc(col.name)}')">${iconSort(col.name)}</button>
          <button class="dt-filt-btn${hasActiveFilter(col.name) ? ' on' : ''}" title="Filter"
            onclick="event.stopPropagation();SarthiDatatable._openFP('${esc(col.name)}',this)">${iconFilter(hasActiveFilter(col.name))}</button>
        </div></th>`;
    });
    thead.innerHTML = html + '</tr>';
  }

  /* ── STATS STRIP ── */
  function renderStatsStrip() {
    const strip = document.getElementById('sdt-stats-strip');
    if (!strip) return;
    Object.values(_st.sparkCharts).forEach(c => { try { c.destroy(); } catch {} });
    _st.sparkCharts = {};

    let html = '<div style="display:flex;width:max-content;">';
    _st.columns.forEach((col, ci) => {
      const w = _st.colWidths[ci];
      if (col.type === 'numeric') {
        const s = col.stats;
        html += `<div class="dt-sc" style="width:${w}px;min-width:${w}px;max-width:${w}px">
          <div class="dt-smr">
            <span class="dt-mv teal" title="Min">↓${fmt(s.min)}</span>
            <span class="dt-mv blue" title="Max">↑${fmt(s.max)}</span>
            ${s.nullCount ? `<span class="dt-mv muted">${s.nullCount}∅</span>` : ''}
          </div>
          <div class="dt-sw" id="sdt-sw-${ci}">
            <canvas id="sdt-sp-${ci}" style="display:block;width:100%;height:100%"></canvas>
            <div class="dt-spk-tt" id="sdt-tt-${ci}"></div>
          </div>
          <div class="dt-sl"><span>${esc(fmt(s.min))}</span><span>${esc(fmt(s.max))}</span></div>
        </div>`;
      } else {
        const s = col.stats;
        const t = (col.uniqueVals || []).slice(0, 2).map(u => esc(u.val)).join(', ');
        html += `<div class="dt-sc" style="width:${w}px;min-width:${w}px;max-width:${w}px">
          <div class="dt-smr"><span class="dt-mv purple">${s.uniqueCount} unique</span>${s.nullCount ? `<span class="dt-mv muted">${s.nullCount}∅</span>` : ''}</div>
          <div class="dt-stop3" title="${t}">${t || '—'}</div>
        </div>`;
      }
    });
    html += '</div>';
    strip.innerHTML = html;

    requestAnimationFrame(() => {
      _st.columns.forEach((col, ci) => { if (col.type === 'numeric') drawSpark(col, ci); });
    });
  }

  /* ── SPARKLINE with dotted mean/median lines ── */
  function drawSpark(col, ci) {
    if (!global.Chart) return;
    const canvas = document.getElementById(`sdt-sp-${ci}`);
    if (!canvas) return;
    const s = col.stats, h = s.hist;
    if (!h.counts || !h.counts.length) return;

    const mn    = s.min, mx = s.max, range = mx - mn || 1;
    const bins  = h.counts.length;
    const meanX = ((s.mean   - mn) / range) * (bins - 1);
    const medX  = ((s.median - mn) / range) * (bins - 1);

    const annotPlugin = {
      id: 'sdtA',
      afterDraw(chart) {
        const ctx = chart.ctx, xs = chart.scales.x, ys = chart.scales.y;
        [[meanX, 'rgba(91,163,245,0.9)'], [medX, 'rgba(112,102,212,0.9)']].forEach(([xv, color]) => {
          const px = xs.getPixelForValue(xv);
          ctx.save(); ctx.beginPath(); ctx.setLineDash([2.5, 2.5]);
          ctx.strokeStyle = color; ctx.lineWidth = 1.5;
          ctx.moveTo(px, ys.top); ctx.lineTo(px, ys.bottom);
          ctx.stroke(); ctx.restore();
        });
      },
    };

    const hoverPlugin = {
      id: 'sdtH',
      afterEvent(chart, args) {
        const tip = document.getElementById(`sdt-tt-${ci}`);
        const sw  = document.getElementById(`sdt-sw-${ci}`);
        if (!tip || !sw) return;
        const ev  = args.event;
        if (ev.type === 'mouseleave') { tip.style.display = 'none'; return; }
        if (ev.type !== 'mousemove')  return;
        const pts = chart.getElementsAtEventForMode(ev.native, 'index', { intersect: false }, true);
        if (!pts.length) { tip.style.display = 'none'; return; }
        const idx   = pts[0].index;
        const binLo = fmt(h.labels[idx]);
        const binHi = fmt(h.labels[idx + 1] ?? s.max);
        tip.innerHTML = `
          <div style="font-size:10px;font-weight:600;color:#fff;margin-bottom:1px;">${binLo} – ${binHi}</div>
          <div style="font-size:9.5px;color:#C8E2FF;">${h.counts[idx]} values</div>
          <div style="font-size:9px;margin-top:3px;">
            <span style="color:rgba(91,163,245,0.95);">— mean ${fmt(s.mean)}</span><br>
            <span style="color:rgba(180,170,240,0.95);">— median ${fmt(s.median)}</span>
          </div>`;
        tip.style.display = 'block';
        const wr  = sw.getBoundingClientRect();
        const ex  = ev.native.clientX - wr.left;
        tip.style.left = (ex > wr.width / 2 ? ex - 125 : ex + 6) + 'px';
        tip.style.top  = '0px';
      },
    };

    try {
      _st.sparkCharts[ci] = new global.Chart(canvas, {
        type: 'bar', plugins: [annotPlugin, hoverPlugin],
        data: {
          labels:   h.counts.map((_, i) => i),
          datasets: [{ data: h.counts, backgroundColor: 'rgba(91,163,245,0.35)',
            borderColor: 'rgba(41,121,217,0.5)', borderWidth: 1,
            borderRadius: 2, borderSkipped: false }],
        },
        options: {
          responsive: true, maintainAspectRatio: false, animation: false,
          events: ['mousemove', 'mouseleave'],
          plugins: { legend: { display: false }, tooltip: { enabled: false } },
          scales: { x: { display: false }, y: { display: false, beginAtZero: true } },
          layout: { padding: { top: 1, bottom: 0, left: 0, right: 0 } },
        },
      });
    } catch (err) { console.warn('spark err', err); }
  }

  /* ── FULL BUILD (called once after data loads) ── */
  function buildFull() {
    computeColWidths();
    renderStatsStrip();
    renderHeader();
    const tbody = document.getElementById('sdt-tbody');
    const cont  = document.getElementById('sdt-table-container');
    if (tbody && cont) renderVirtualRows(tbody, cont);
    renderChips(); updateStatus();
    // Sync horizontal scroll
    if (cont) {
      cont.addEventListener('scroll', () => {
        const strip = document.getElementById('sdt-stats-strip');
        if (strip) strip.scrollLeft = cont.scrollLeft;
        const tb = document.getElementById('sdt-tbody');
        if (tb) renderVirtualRows(tb, cont);
      }, { passive: true });
    }
  }

  function refresh() {
    applyFilters(); renderHeader();
    const tbody = document.getElementById('sdt-tbody');
    const cont  = document.getElementById('sdt-table-container');
    if (tbody && cont) renderVirtualRows(tbody, cont);
    renderChips(); updateStatus();
  }

  /* ── FILTER PANEL ── */
  function openFP(colName, btn) {
    const col   = _st.columns.find(c => c.name === colName);
    const panel = document.getElementById('sdt-fp');
    if (!col || !panel) return;
    if (_st.panelOpen === colName) { closeFP(); return; }
    _st.panelOpen = colName;
    const r  = btn.getBoundingClientRect();
    panel.style.left    = Math.min(r.left, window.innerWidth - 265) + 'px';
    panel.style.top     = (r.bottom + 5) + 'px';
    panel.style.display = 'block';
    col.type === 'string' ? renderSFP(col, panel) : renderNFP(col, panel);
    setTimeout(() => document.addEventListener('click', outsideFP, { once: true }), 10);
  }
  function outsideFP(e) {
    const p = document.getElementById('sdt-fp');
    if (p && !p.contains(e.target)) closeFP();
  }
  function closeFP() {
    const p = document.getElementById('sdt-fp');
    if (p) p.style.display = 'none';
    _st.panelOpen = null;
  }

    const tx = f.text || '', sel = f.selected || [];
    const all = col.uniqueVals.slice(0, 300);
    const vis = tx ? all.filter(u => u.val.toLowerCase().includes(tx.toLowerCase())) : all;
    panel.innerHTML = `
      <div style="padding:8px 12px 6px;border-bottom:1px solid rgba(100,130,180,0.09);display:flex;align-items:center;justify-content:space-between;">
        <span style="font-size:11px;font-weight:600;color:#1A2535;overflow:hidden;text-overflow:ellipsis;white-space:nowrap;max-width:155px">${esc(col.name)}</span>
        <span style="font-size:10px;color:#7A8BA8">${col.stats.uniqueCount} unique</span>
      </div>
      <div style="padding:7px 12px;border-bottom:1px solid rgba(100,130,180,0.09)">
        <div style="font-size:9.5px;font-weight:700;letter-spacing:.5px;text-transform:uppercase;color:#7A8BA8;margin-bottom:4px">Search</div>
        <input style="width:100%;padding:5px 8px;font-size:11.5px;font-family:'DM Sans',sans-serif;border:1.5px solid rgba(100,130,180,0.22);border-radius:7px;background:#F7F9FC;color:#1A2535;outline:none;box-sizing:border-box"
          id="sdt-ss" value="${esc(tx)}" placeholder="Type to filter…"
          onfocus="this.style.borderColor='#5BA3F5'" onblur="this.style.borderColor='rgba(100,130,180,0.22)'"
          oninput="SarthiDatatable._sSC('${esc(col.name)}',this.value)"/>
      </div>
      <div style="max-height:190px;overflow-y:auto;padding:4px 8px">
        ${sel.length ? `<div style="display:flex;justify-content:space-between;padding:2px 4px 4px;font-size:10px;color:#7A8BA8">
          <span>${sel.length} selected</span>
          <button onclick="SarthiDatatable._clrC('${esc(col.name)}')" style="color:#5BA3F5;background:none;border:none;cursor:pointer;font-family:'DM Sans',sans-serif;font-size:10px">Clear</button>
        </div>` : ''}
        ${vis.map(u => `<div style="display:flex;align-items:center;gap:7px;padding:4px 6px;cursor:pointer;border-radius:5px;font-size:12px;color:#1A2535;transition:background .09s"
          onmouseover="this.style.background='#F0F4FF'" onmouseout="this.style.background=''">
          <input type="checkbox" ${sel.includes(u.val) ? 'checked' : ''} style="accent-color:#2979D9;width:12px;height:12px;flex-shrink:0"
            onchange="SarthiDatatable._tVF('${esc(col.name)}','${esc(u.val)}')"/>
          <span style="flex:1;overflow:hidden;text-overflow:ellipsis;white-space:nowrap" title="${esc(u.val)}">${esc(u.val)}</span>
          <span style="font-size:9.5px;color:#7A8BA8;flex-shrink:0">${u.count}</span>
        </div>`).join('')}
        ${!vis.length ? '<div style="text-align:center;padding:10px;font-size:11.5px;color:#7A8BA8">No matches</div>' : ''}
      </div>`;
  }

  /* ── CHIPS ── */
  function renderChips() {
    const el = document.getElementById('sdt-chips');
    if (!el) return;

    Object.entries(_st.filters).forEach(([cn, f]) => {
      const col = _st.columns.find(c => c.name === cn); if (!col) return;
      const clr = `SarthiDatatable._clrC('${esc(cn)}')`;
      if (col.type === 'string') {
        if (f.selected && f.selected.length) h += mkChip(`${esc(cn)}: ${f.selected.length === 1 ? esc(f.selected[0]) : f.selected.length + ' values'}`, clr);
        if (f.text) h += mkChip(`${esc(cn)} ∋ "${esc(f.text)}"`, clr);
      } else {
        if (f.min !== '' && f.min != null) h += mkChip(`${esc(cn)} ≥ ${fmt(f.min)}`, clr);
        if (f.max !== '' && f.max != null) h += mkChip(`${esc(cn)} ≤ ${fmt(f.max)}`, clr);
      }
    });
    el.innerHTML = h;
  }
  function mkChip(label, clrFn) {

  }

  function hasActiveFilter(cn) {
    const f = _st.filters[cn]; if (!f) return false;
    return !!(f.selected && f.selected.length) || !!f.text ||
           (f.min !== '' && f.min != null) || (f.max !== '' && f.max != null);
  }
  function fmt(n) {
    if (n === undefined || n === null) return '—';
    const v = parseFloat(n); if (isNaN(v)) return '—';
    if (Math.abs(v) >= 1e6)  return (v/1e6).toFixed(2)+'M';
    if (Math.abs(v) >= 1e3)  return (v/1e3).toFixed(1)+'K';
    return Number.isInteger(v) ? v.toLocaleString() : v.toFixed(2);
  }
  function esc(s) {
    return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
                        .replace(/"/g,'&quot;').replace(/'/g,'&#39;');
  }

  const SarthiDatatable = {
    open(csvUrl, title) {
      Object.assign(_st, { csvUrl, title: title||'Data Table', filters:{}, sortCol:null, sortDir:'none', globalSearch:'', panelOpen:null });
      if (!document.getElementById('sdt-overlay')) {
        const d = document.createElement('div'); d.innerHTML = buildModalHTML(); document.body.appendChild(d);
      }
      document.getElementById('sdt-overlay').style.display = 'flex';
      document.body.style.overflow = 'hidden';
      const te = document.getElementById('sdt-title'); if (te) te.textContent = title || 'Data Table';
      showLoading('Loading CSV…');
      ensureChartJS(() => {
        fetch(csvUrl)
          .then(r => { if (!r.ok) throw new Error(`HTTP ${r.status}`); return r.text(); })
          .then(text => {
            const { headers, rows } = parseCSV(text);
            _st.rawData = rows; _st.columns = detectColumns(headers, rows); _st.filteredData = rows; _st.filters = {};
            buildFull();
            const gs = document.getElementById('sdt-gs');
            if (gs) { gs.value = ''; gs.oninput = e => { _st.globalSearch = e.target.value; refresh(); }; }
          })
          .catch(err => showLoading(`Failed: ${err.message}`));
      });
    },

    close() {
      const o = document.getElementById('sdt-overlay'); if (o) o.style.display = 'none';
      document.body.style.overflow = '';
      closeFP();
      Object.values(_st.sparkCharts).forEach(c => { try { c.destroy(); } catch {} });
      _st.sparkCharts = {};
    },

    _sortBy(cn) {
      if (_st.sortCol === cn) {
        _st.sortDir = _st.sortDir === 'none' ? 'asc' : _st.sortDir === 'asc' ? 'desc' : 'none';
        if (_st.sortDir === 'none') _st.sortCol = null;
      } else { _st.sortCol = cn; _st.sortDir = 'asc'; }
      refresh();
      const c = document.getElementById('sdt-table-container'); if (c) c.scrollTop = 0;
    },

    _openFP(cn, btn)      { openFP(cn, btn); },
    _tVF(cn, val)         {
      if (!_st.filters[cn]) _st.filters[cn] = { selected:[], text:'' };
      const s = _st.filters[cn].selected, i = s.indexOf(val);
      i === -1 ? s.push(val) : s.splice(i,1);
      if (!s.length && !_st.filters[cn].text) delete _st.filters[cn];
      refresh();
      const col = _st.columns.find(c=>c.name===cn), p = document.getElementById('sdt-fp');
      if (col && p && p.style.display!=='none') { renderSFP(col,p); const ss=document.getElementById('sdt-ss'); if(ss) ss.focus(); }
    },
    _sSC(cn, val)         {
      if (!_st.filters[cn]) _st.filters[cn] = { selected:[], text:'' };
      _st.filters[cn].text = val;
      if (!val && !(_st.filters[cn].selected||[]).length) delete _st.filters[cn];
      refresh();
      const col=_st.columns.find(c=>c.name===cn), p=document.getElementById('sdt-fp');
      if (col && p && p.style.display!=='none') { renderSFP(col,p); const ss=document.getElementById('sdt-ss'); if(ss){ss.focus();ss.value=val;} }
    },
    _nFC(cn)              {
      const mn=document.getElementById('sdt-nm'), mx=document.getElementById('sdt-nx');
      if (!_st.filters[cn]) _st.filters[cn] = {};
      _st.filters[cn].min = mn?mn.value:''; _st.filters[cn].max = mx?mx.value:'';
      if (!_st.filters[cn].min && !_st.filters[cn].max) delete _st.filters[cn];
      refresh();
    },
    _clrC(cn)             { delete _st.filters[cn]; refresh(); closeFP(); },
    _clrG()               { _st.globalSearch=''; const g=document.getElementById('sdt-gs'); if(g) g.value=''; refresh(); },
    _resetAll()           {
