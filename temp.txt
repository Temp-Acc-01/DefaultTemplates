_state.colWidths    = computeColWidths(_state.columns, rows);

            refresh();

            // Wire global search
            const gs = document.getElementById('sdt-global-search');
            if (gs) {
              gs.value = '';
              gs.oninput = (e) => { _state.globalSearch = e.target.value; refresh(); };
            }

            // Wire virtual scroll + ResizeObserver
            const cont = document.getElementById('sdt-table-container');
            if (cont) {
              cont.onscroll = () => {
                _state.scrollTop = cont.scrollTop;
                const tbody = document.getElementById('sdt-tbody');
                if (tbody) renderVirtualRows(tbody, cont);
              };

              if (_state._resizeObserver) _state._resizeObserver.disconnect();
              _state._resizeObserver = new ResizeObserver(() => {
                _state.colWidths = computeColWidths(_state.columns, _state.rawData);
                refresh();
              });
              _state._resizeObserver.observe(cont);
            }
