  /* ═══════════════════════════════════════════════════════════════════
     CSV PARSER
  ═══════════════════════════════════════════════════════════════════ */
  function parseCSV(text) {
    const lines = text.split(/\r?\n/);
    if (lines.length < 2) return { headers: [], rows: [] };
    const headers = splitCSVLine(lines[0]);
    const rows = [];
    for (let i = 1; i < lines.length; i++) {
      if (!lines[i].trim()) continue;
      const vals = splitCSVLine(lines[i]);
      const obj = {};
      headers.forEach((h, j) => { obj[h] = vals[j] ?? ''; });
      rows.push(obj);
    }
    return { headers, rows };
  }

  function splitCSVLine(line) {
    const result = [];
    let cur = '', inQ = false;
    for (let i = 0; i < line.length; i++) {
      const ch = line[i];
      if (ch === '"') { inQ = !inQ; continue; }
      if (ch === ',' && !inQ) { result.push(cur.trim()); cur = ''; continue; }
      cur += ch;
    }
    result.push(cur.trim());
    return result;
  }

  /* ═══════════════════════════════════════════════════════════════════
     TYPE DETECTION & STATS
  ═══════════════════════════════════════════════════════════════════ */
  function detectColumns(headers, rows) {
    return headers.map(name => {
      const vals = rows.map(r => r[name]).filter(v => v !== '' && v !== null && v !== undefined);
      const numVals = vals.map(v => parseFloat(v)).filter(v => !isNaN(v));
      const isNumeric = numVals.length / Math.max(vals.length, 1) > 0.85;

      if (isNumeric) {
        numVals.sort((a, b) => a - b);
        const min    = numVals[0] ?? 0;
        const max    = numVals[numVals.length - 1] ?? 0;
        const mean   = numVals.reduce((a, b) => a + b, 0) / (numVals.length || 1);
        const mid    = Math.floor(numVals.length / 2);
        const median = numVals.length % 2 === 0
          ? (numVals[mid - 1] + numVals[mid]) / 2
          : numVals[mid];
        const nullCount = rows.length - vals.length;
        const hist = buildHistogram(numVals, 10);
        return { name, type: 'numeric', stats: { min, max, mean, median, nullCount, hist }, uniqueVals: [] };
      } else {
        const freq = {};
        vals.forEach(v => { freq[v] = (freq[v] || 0) + 1; });
        const uniqueVals = Object.entries(freq)
          .sort((a, b) => b[1] - a[1])
          .map(([val, count]) => ({ val, count }));
        const nullCount = rows.length - vals.length;
        return { name, type: 'string', stats: { uniqueCount: uniqueVals.length, nullCount }, uniqueVals };
      }
    });
  }

  // Compute stats on currently filtered data for a numeric column
  function computeFilteredStats(colName, filteredRows) {
    const numVals = filteredRows
      .map(r => parseFloat(r[colName]))
      .filter(v => !isNaN(v));
    if (!numVals.length) return null;
    numVals.sort((a, b) => a - b);
    const min    = numVals[0];
    const max    = numVals[numVals.length - 1];
    const mean   = numVals.reduce((a, b) => a + b, 0) / numVals.length;
    const mid    = Math.floor(numVals.length / 2);
    const median = numVals.length % 2 === 0
      ? (numVals[mid - 1] + numVals[mid]) / 2
      : numVals[mid];
    const nullCount = filteredRows.length - numVals.length;
    const hist = buildHistogram(numVals, 10);
    return { min, max, mean, median, nullCount, hist };
  }

  function buildHistogram(sortedNums, bins) {
    if (!sortedNums.length) return { bins: [], counts: [], labels: [] };
    const min = sortedNums[0], max = sortedNums[sortedNums.length - 1];
    const step = (max - min) / bins || 1;
    const counts = new Array(bins).fill(0);
    const labels = [];
    for (let i = 0; i < bins; i++) labels.push(+(min + i * step).toFixed(2));
    sortedNums.forEach(v => {
      const idx = Math.min(Math.floor((v - min) / step), bins - 1);
      counts[idx]++;
    });
    return { labels, counts, step, min, max };
  }

  /* ═══════════════════════════════════════════════════════════════════
     COLUMN WIDTHS  (shared between stats strip & table)
  ═══════════════════════════════════════════════════════════════════ */
  function computeColWidths(columns, sampleRows) {
    return columns.map(col => {
      // Estimate based on header length + sample values
      let maxLen = col.name.length;
      sampleRows.slice(0, 20).forEach(r => {
        const v = String(r[col.name] || '');
        if (v.length > maxLen) maxLen = v.length;
      });
      // ~7.5px per char, capped
      return Math.max(COL_MIN_W, Math.min(maxLen * 7.5 + COL_PAD, 260));
    });
  }

  /* ═══════════════════════════════════════════════════════════════════
     FILTER ENGINE
  ═══════════════════════════════════════════════════════════════════ */
  function applyFilters() {
    let data = _state.rawData;

    if (_state.globalSearch) {
      const q = _state.globalSearch.toLowerCase();
      data = data.filter(row =>
        Object.values(row).some(v => String(v).toLowerCase().includes(q))
      );
    }

    _state.columns.forEach(col => {
      const f = _state.filters[col.name];
      if (!f) return;
      if (col.type === 'string') {
        if (f.selected && f.selected.length > 0)
          data = data.filter(r => f.selected.includes(r[col.name]));
        if (f.text) {
          const q = f.text.toLowerCase();
          data = data.filter(r => String(r[col.name]).toLowerCase().includes(q));
        }
      } else {
        if (f.min !== undefined && f.min !== '')
          data = data.filter(r => parseFloat(r[col.name]) >= parseFloat(f.min));
        if (f.max !== undefined && f.max !== '')
          data = data.filter(r => parseFloat(r[col.name]) <= parseFloat(f.max));
      }
    });

    if (_state.sortCol) {
      const col = _state.columns.find(c => c.name === _state.sortCol);
      const dir = _state.sortDir === 'asc' ? 1 : -1;
      data = [...data].sort((a, b) => {
        const av = a[_state.sortCol], bv = b[_state.sortCol];
        if (col && col.type === 'numeric')
          return (parseFloat(av) - parseFloat(bv)) * dir;
        return String(av).localeCompare(String(bv)) * dir;
      });
    }

    _state.filteredData = data;
  }

  /* ═══════════════════════════════════════════════════════════════════
     VIRTUAL SCROLL RENDERER
  ═══════════════════════════════════════════════════════════════════ */
  function renderVirtualRows(tableBody, scrollContainer) {
    const data   = _state.filteredData;
    const cols   = _state.columns;
    const viewH  = scrollContainer.clientHeight;
    const scrollT = scrollContainer.scrollTop;

    const startIdx = Math.max(0, Math.floor(scrollT / VIRTUAL_ROW_H) - VIRTUAL_BUFFER);
    const endIdx   = Math.min(data.length, startIdx + Math.ceil(viewH / VIRTUAL_ROW_H) + VIRTUAL_BUFFER * 2);

    const topH = startIdx * VIRTUAL_ROW_H;
    const botH = (data.length - endIdx) * VIRTUAL_ROW_H;

    let html = `<tr style="height:${topH}px"><td colspan="${cols.length}" style="padding:0;border:none"></td></tr>`;

    for (let i = startIdx; i < endIdx; i++) {
      const row = data[i];
      const bg = i % 2 === 0 ? '#fff' : '#F7F9FC';
      html += `<tr class="dt-row" style="background:${bg}">`;
      cols.forEach((col, ci) => {
        const val = row[col.name] ?? '';
        const display = val === '' ? '<span class="dt-null">—</span>' : esc(String(val));
        const w = _state.colWidths[ci] || COL_MIN_W;
        html += `<td class="dt-cell" style="min-width:${w}px;max-width:${w}px;width:${w}px;" title="${esc(String(val))}">${display}</td>`;
      });
      html += '</tr>';
    }

    html += `<tr style="height:${botH}px"><td colspan="${cols.length}" style="padding:0;border:none"></td></tr>`;
    tableBody.innerHTML = html;
  }

        // desc — down arrow solid, up faint
        sortIconSvg = `<svg width="10" height="14" viewBox="0 0 10 14" fill="currentColor">
          <path d="M5 1L2 5h6L5 1z" style="opacity:0.2"/>
          <path d="M5 13l3-4H2l3 4z" style="opacity:1;fill:#2979D9"/>
        </svg>`;
      }

      // Filter icon — horizontal sliders (equalizer style)
      const filterBtnCls = hasFilter ? 'dt-filter-btn active' : 'dt-filter-btn';
      const filterIconSvg = `<svg width="13" height="13" viewBox="0 0 20 20" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linecap="round">
        <line x1="3" y1="5" x2="17" y2="5"/>
        <circle cx="12" cy="5" r="2.2" fill="currentColor" stroke="none" style="opacity:0.9"/>
        <line x1="3" y1="10" x2="17" y2="10"/>
        <circle cx="7" cy="10" r="2.2" fill="currentColor" stroke="none" style="opacity:0.9"/>
        <line x1="3" y1="15" x2="17" y2="15"/>
        <circle cx="13" cy="15" r="2.2" fill="currentColor" stroke="none" style="opacity:0.9"/>
      </svg>`;

      headRow += `<th class="dt-th" style="min-width:${w}px;max-width:${w}px;width:${w}px;" data-col="${esc(col.name)}">
        <div class="dt-th-inner" onclick="SarthiDatatable._sortBy('${esc(col.name)}')">
          <span class="dt-type-badge ${typeCls}">${typeLabel}</span>
          <span style="overflow:hidden;text-overflow:ellipsis;flex:1;max-width:${w - 80}px;" title="${esc(col.name)}">${esc(col.name)}</span>
          <span class="dt-sort-icon ${isSort ? (_state.sortDir === 'asc' ? 'active-asc' : 'active-desc') : ''}">${sortIconSvg}</span>
          <button class="${filterBtnCls}" title="Filter ${esc(col.name)}"
            onclick="event.stopPropagation();SarthiDatatable._openFilterPanel('${esc(col.name)}', this)"
          >${filterIconSvg}</button>
        </div>
      </th>`;
    });
    headRow += '</tr>';

    thead.innerHTML = statsRow + headRow;

    // Draw sparklines after DOM update
    requestAnimationFrame(() => drawAllSparklines());
  }

  /* ═══════════════════════════════════════════════════════════════════
     SPARKLINE DRAWING (Chart.js)
  ═══════════════════════════════════════════════════════════════════ */
  function drawAllSparklines() {
    // Destroy old
    Object.values(_state.sparkCharts).forEach(c => { try { c.destroy(); } catch {} });
    _state.sparkCharts = {};

    if (!global.Chart) return;

    _state.columns.forEach((col, ci) => {
      if (col.type !== 'numeric') return;

      const canvas = document.getElementById(`sdt-spark-${ci}`);
      const wrap   = document.getElementById(`sdt-sparkwrap-${ci}`);
      const meanEl = document.getElementById(`sdt-mean-${ci}`);
      const medEl  = document.getElementById(`sdt-med-${ci}`);
      if (!canvas || !wrap) return;

      // Use filtered data stats when filters are active
      const s = _state.filteredData.length < _state.rawData.length
        ? (computeFilteredStats(col.name, _state.filteredData) || col.stats)
        : col.stats;
      const hist = s.hist;
      if (!hist || !hist.counts || !hist.counts.length) return;

      const colName = col.name;

      try {
        _state.sparkCharts[ci] = new global.Chart(canvas, {
          type: 'bar',
          data: {
            labels: hist.labels,
            datasets: [{
              data: hist.counts,
              backgroundColor: 'rgba(91,163,245,0.45)',
              borderColor:     'rgba(41,121,217,0.55)',
              borderWidth: 1,
              borderRadius: 2,
              hoverBackgroundColor: 'rgba(41,121,217,0.7)',
            }]
          },
          options: {
            responsive:          false,
            animation:           false,
            plugins: {
              legend:  { display: false },
              tooltip: {
                enabled: true,
                callbacks: {
                  title: (items) => {
                    const idx = items[0].dataIndex;
                    const lo  = hist.labels[idx];
                    const hi  = +(lo + (hist.step || 0)).toFixed(2);
                    return `${fmt(lo)} – ${fmt(hi)}`;
                  },
                  label: (item) => `Count: ${item.raw}`,
                },
                backgroundColor: '#1A2535',
                titleColor:      '#fff',
                bodyColor:       '#C8E2FF',
                padding:         6,
                cornerRadius:    6,
                displayColors:   false,
              }
            },
            scales: {
              x: { display: false },
              y: { display: false, beginAtZero: true }
            },
            layout:              { padding: 0 },
            onClick: (evt, elements) => {
              if (!elements.length) return;
              const idx = elements[0].index;
              const lo  = hist.labels[idx];
              const hi  = +(lo + (hist.step || 1)).toFixed(4);
              // Set range filter for this bar
              if (!_state.filters[colName]) _state.filters[colName] = {};
              _state.filters[colName].min = lo;
              _state.filters[colName].max = hi;
              refresh();
            },
          }
        });

        // Position mean & median dashed lines
        positionMarker(meanEl, medEl, s, hist, wrap);

      } catch (e) { /* silent */ }
    });
  }

  function positionMarker(meanEl, medEl, s, hist, wrap) {
    if (!meanEl || !medEl || !wrap) return;
    const wrapW = wrap.offsetWidth || 120;
    const range = hist.max - hist.min;
    if (range <= 0) return;

    const meanPct   = (s.mean   - hist.min) / range;
    const medianPct = (s.median - hist.min) / range;

    meanEl.style.left   = Math.min(Math.max(meanPct   * wrapW, 1), wrapW - 2) + 'px';
    medEl.style.left    = Math.min(Math.max(medianPct * wrapW, 1), wrapW - 2) + 'px';
  }

  /* ═══════════════════════════════════════════════════════════════════
     FILTER PANEL
  ═══════════════════════════════════════════════════════════════════ */
  function openFilterPanel(colName, btnEl) {
    const col   = _state.columns.find(c => c.name === colName);
    const panel = document.getElementById('sdt-filter-panel');
    if (!col || !panel) return;

    if (_state.panelOpen === colName) { closeFilterPanel(); return; }
    _state.panelOpen = colName;

    const rect = btnEl.getBoundingClientRect();
    panel.style.left    = Math.min(rect.left, window.innerWidth - 260) + 'px';
    panel.style.top     = (rect.bottom + 4) + 'px';
    panel.style.display = 'block';

    if (col.type === 'string') renderStringFilterPanel(col, panel);
    else renderNumericFilterPanel(col, panel);

    setTimeout(() => {
      document.addEventListener('click', _outsideFilterClose, { once: true });
    }, 10);
  }

  function _outsideFilterClose(e) {
    const panel = document.getElementById('sdt-filter-panel');
    if (panel && !panel.contains(e.target)) closeFilterPanel();
  }

    if (panel) panel.style.display = 'none';
    _state.panelOpen = null;
  }

  /* ═══════════════════════════════════════════════════════════════════
     FILTER CHIPS
  ═══════════════════════════════════════════════════════════════════ */
  function renderChips() {
    const chips    = document.getElementById('sdt-chips');
    const hasAny   = Object.keys(_state.filters).some(k => hasActiveFilter(k));
    const hasGlob  = !!_state.globalSearch;
    if (!chips) return;
    if (!hasAny && !hasGlob) { chips.style.display = 'none'; chips.innerHTML = ''; return; }
    chips.style.display = 'flex';

    let html = '';
    if (hasGlob) {
      html += `<span class="sdt-chip">Search: "${esc(_state.globalSearch)}"
        <button class="sdt-chip-x" onclick="SarthiDatatable._clearGlobal()">×</button></span>`;
    }
    Object.entries(_state.filters).forEach(([col, f]) => {
      const cm = _state.columns.find(c => c.name === col);
      if (!cm) return;
      if (cm.type === 'string') {
        if (f.selected && f.selected.length) {
          const lbl = f.selected.length === 1 ? esc(f.selected[0]) : `${f.selected.length} values`;
          html += `<span class="sdt-chip">${esc(col)}: ${lbl}
            <button class="sdt-chip-x" onclick="SarthiDatatable._clearColFilter('${esc(col)}')">×</button></span>`;
        }
        if (f.text) {
          html += `<span class="sdt-chip">${esc(col)} contains "${esc(f.text)}"
            <button class="sdt-chip-x" onclick="SarthiDatatable._clearColFilter('${esc(col)}')">×</button></span>`;
        }
      } else {
        if (f.min !== '' && f.min !== undefined)
          html += `<span class="sdt-chip">${esc(col)} ≥ ${fmt(f.min)}
            <button class="sdt-chip-x" onclick="SarthiDatatable._clearColFilter('${esc(col)}')">×</button></span>`;
        if (f.max !== '' && f.max !== undefined)
          html += `<span class="sdt-chip">${esc(col)} ≤ ${fmt(f.max)}
            <button class="sdt-chip-x" onclick="SarthiDatatable._clearColFilter('${esc(col)}')">×</button></span>`;
      }
    });
    chips.innerHTML = html;
  }

  /* ═══════════════════════════════════════════════════════════════════
     STATUS
  ═══════════════════════════════════════════════════════════════════ */
  function updateStatus() {
    const left  = document.getElementById('sdt-status-left');
    const count = document.getElementById('sdt-row-count');
    const total = _state.rawData.length;
    const shown = _state.filteredData.length;
    const txt   = shown < total
      ? `${shown.toLocaleString()} of ${total.toLocaleString()} rows · ${_state.columns.length} columns`
      : `${total.toLocaleString()} rows · ${_state.columns.length} columns`;
    if (left)  left.textContent  = txt;
    if (count) count.textContent = txt;
  }

  /* ═══════════════════════════════════════════════════════════════════
     CHART.JS LOADER
  ═══════════════════════════════════════════════════════════════════ */
  function ensureChartJS(cb) {
    if (global.Chart) { cb(); return; }
    const s = document.createElement('script');
    s.src = CHARTJS_CDN;
    s.onload  = cb;
    s.onerror = cb;
    document.head.appendChild(s);
  }

  /* ═══════════════════════════════════════════════════════════════════
     PUBLIC API
  ═══════════════════════════════════════════════════════════════════ */
  const SarthiDatatable = {

    open(csvUrl, title) {
      _state.csvUrl       = csvUrl;
      _state.title        = title || 'Data Table';
      _state.filters      = {};
      _state.sortCol      = null;
      _state.sortDir      = 'asc';
      _state.globalSearch = '';
      _state.panelOpen    = null;
      _state.sparkCharts  = {};

      if (!document.getElementById('sdt-overlay')) {
        const div = document.createElement('div');
        div.innerHTML = buildModalHTML();
        document.body.appendChild(div);
      }

      const overlay = document.getElementById('sdt-overlay');
      overlay.style.display = 'flex';
      document.body.style.overflow = 'hidden';

      const titleEl = document.getElementById('sdt-title');
      if (titleEl) titleEl.textContent = title || 'Data Table';

      showLoading('Loading data…');

      ensureChartJS(() => {
        fetch(csvUrl)
          .then(r => { if (!r.ok) throw new Error(`HTTP ${r.status}`); return r.text(); })
          .then(text => {
            const { headers, rows } = parseCSV(text);
            _state.rawData      = rows;
            _state.columns      = detectColumns(headers, rows);
            _state.filteredData = rows;
            _state.filters      = {};
            _state.colWidths    = computeColWidths(_state.columns, rows);

            refresh();

            // Wire global search
            const gs = document.getElementById('sdt-global-search');
            if (gs) {
              gs.value = '';
              gs.oninput = (e) => { _state.globalSearch = e.target.value; refresh(); };
            }

            // Wire virtual scroll
            const cont = document.getElementById('sdt-table-container');
            if (cont) {
              cont.onscroll = () => {
                _state.scrollTop = cont.scrollTop;
                const tbody = document.getElementById('sdt-tbody');
                if (tbody) renderVirtualRows(tbody, cont);
              };
            }
          })
          .catch(err => { showLoading(`Failed to load data: ${err.message}`); });
      });
    },

    close() {
      const overlay = document.getElementById('sdt-overlay');
      if (overlay) overlay.style.display = 'none';
      document.body.style.overflow = '';
      closeFilterPanel();
      Object.values(_state.sparkCharts).forEach(c => { try { c.destroy(); } catch {} });
      _state.sparkCharts = {};
    },

    _sortBy(colName) {
      if (_state.sortCol === colName) {
        _state.sortDir = _state.sortDir === 'asc' ? 'desc' : 'asc';
      } else {
        _state.sortCol = colName;
        _state.sortDir = 'asc';
      }
      refresh();
      const cont = document.getElementById('sdt-table-container');
      if (cont) cont.scrollTop = 0;
    },

    _openFilterPanel(colName, btnEl) { openFilterPanel(colName, btnEl); },

    _toggleValFilter(colName, val) {
      if (!_state.filters[colName]) _state.filters[colName] = { selected: [], text: '' };
      const sel = _state.filters[colName].selected;
      const idx = sel.indexOf(val);
      if (idx === -1) sel.push(val); else sel.splice(idx, 1);
      if (!sel.length && !_state.filters[colName].text) delete _state.filters[colName];
      refresh();
      const col   = _state.columns.find(c => c.name === colName);
      const panel = document.getElementById('sdt-filter-panel');
      if (col && panel && panel.style.display !== 'none') {
        renderStringFilterPanel(col, panel);
        const si = document.getElementById('sdt-str-search');
        if (si) si.focus();
      }
    },

    _strSearchChange(colName, val) {
      if (!_state.filters[colName]) _state.filters[colName] = { selected: [], text: '' };
      _state.filters[colName].text = val;
      if (!val && !(_state.filters[colName].selected || []).length) delete _state.filters[colName];
      refresh();
      const col   = _state.columns.find(c => c.name === colName);
      const panel = document.getElementById('sdt-filter-panel');
      if (col && panel && panel.style.display !== 'none') {
        renderStringFilterPanel(col, panel);
        const si = document.getElementById('sdt-str-search');
        if (si) { si.focus(); si.value = val; }
      }
    },

    _numFilterChange(colName) {
      const minEl = document.getElementById('sdt-num-min');
      const maxEl = document.getElementById('sdt-num-max');
      if (!_state.filters[colName]) _state.filters[colName] = {};
      _state.filters[colName].min = minEl ? minEl.value : '';
      _state.filters[colName].max = maxEl ? maxEl.value : '';
      if (!_state.filters[colName].min && !_state.filters[colName].max)
        delete _state.filters[colName];
      refresh();
    },

    _clearColFilter(colName) {
      delete _state.filters[colName];
      refresh();
      closeFilterPanel();
    },

    _clearGlobal() {
      _state.globalSearch = '';
      const gs = document.getElementById('sdt-global-search');
      if (gs) gs.value = '';
      refresh();
    },

    _resetAll() {
      _state.filters      = {};
      _state.globalSearch = '';
      _state.sortCol      = null;
      _state.sortDir      = 'asc';
      const gs = document.getElementById('sdt-global-search');
      if (gs) gs.value = '';
      closeFilterPanel();
      refresh();
      const cont = document.getElementById('sdt-table-container');
      if (cont) cont.scrollTop = 0;
    },
  };

  /* ═══════════════════════════════════════════════════════════════════
     HELPERS
  ═══════════════════════════════════════════════════════════════════ */
  function hasActiveFilter(colName) {
    const f = _state.filters[colName];
    if (!f) return false;
    if (f.selected && f.selected.length > 0) return true;
    if (f.text) return true;
    if (f.min !== '' && f.min !== undefined) return true;
    if (f.max !== '' && f.max !== undefined) return true;
    return false;
  }

  function fmt(n) {
    if (n === undefined || n === null || isNaN(n)) return '—';
    const num = parseFloat(n);
    if (Math.abs(num) >= 1e6)   return (num / 1e6).toFixed(2)  + 'M';
    if (Math.abs(num) >= 1e3)   return (num / 1e3).toFixed(1)  + 'K';
    if (Number.isInteger(num))  return num.toLocaleString();
    return num.toFixed(2);
  }

  function esc(str) {
    if (!str) return '';
    return String(str)
      .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
  }
