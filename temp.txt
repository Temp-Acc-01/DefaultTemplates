/* ═══════════════════════════════════════════════════════════════
   TIMESTAMP
═══════════════════════════════════════════════════════════════ */
function nowTs() {
  const d = new Date(), pad = n => String(n).padStart(2,'0');
  return `${pad(d.getHours())}:${pad(d.getMinutes())}:${pad(d.getSeconds())}`;
}
function tsEl() {
  const s = document.createElement('span');
  s.className = 'msg-ts'; s.textContent = nowTs(); return s;
}

/* ═══════════════════════════════════════════════════════════════
   SESSION MANAGEMENT
═══════════════════════════════════════════════════════════════ */
function newChat() {
  const s = { id: Date.now(), title: 'New conversation', messages: [] };
  sessions.unshift(s);
  currentSession = s;
  renderHistory();
  renderChat();
  document.getElementById('session-title').textContent = s.title;
  if (!isDesktop()) closeSidebar();
}

function renderHistory() {
  document.getElementById('history-list').innerHTML = sessions.map(s => `
    <div class="history-item ${s.id === currentSession?.id ? 'active' : ''}"
         onclick="switchSession(${s.id})">
      <svg viewBox="0 0 24 24" fill="currentColor"><path d="M20 2H4c-1.1 0-2 .9-2 2v18l4-4h14c1.1 0 2-.9 2-2V4c0-1.1-.9-2-2-2z"/></svg>
      ${escHtml(s.title)}
    </div>`).join('');
}

function switchSession(id) {
  currentSession = sessions.find(s => s.id === id);
  renderHistory();
  renderChat();
  document.getElementById('session-title').textContent = currentSession.title;
  if (!isDesktop()) closeSidebar();
}

function renderChat() {
  const area = document.getElementById('chat-area');
  if (!currentSession || currentSession.messages.length === 0) {
    area.innerHTML = `<div class="welcome" id="welcome-screen">
      <h1>Ask your data anything</h1>
      <p>Sarthi translates your questions into insights — charts, trends, and answers from your database.</p>
    </div>`;
    return;
  }
  area.innerHTML = '';
  currentSession.messages.forEach(m => area.appendChild(buildStaticMessage(m)));
  scrollToBottom();
}

function fmtTokens(n) {
  if (n === null || n === undefined) return '—';
  if (n >= 1_000_000) return (n / 1_000_000).toFixed(2) + 'M';
  if (n >= 1_000)     return (n / 1_000).toFixed(1) + 'K';
  return n.toLocaleString();
}

function updateTokenUI() {
  const wrap = document.getElementById('token-balance-wrap');
  const pill = document.getElementById('token-pill');
  const txt  = document.getElementById('token-text');
  if (!wrap) return;

  if (tokenState.available === null && !tokenState.unlimited) { wrap.style.display = 'none'; return; }
  wrap.style.display = 'flex';

  pill.className = 'token-pill' + (tokenState.warn ? ' warn' : '') + (tokenState.blocked ? ' blocked' : '');
  txt.textContent = tokenState.unlimited ? '∞' : fmtTokens(tokenState.available);

  const tibAlloc = document.getElementById('tib-allocated');
  const tibUsed  = document.getElementById('tib-used');
  const tibAvail = document.getElementById('tib-available');
  if (tibAlloc) tibAlloc.textContent = (tokenState.total ?? 0).toLocaleString();
  if (tibUsed)  tibUsed.textContent  = ((tokenState.total ?? 0) - (tokenState.available ?? 0)).toLocaleString();
  if (tibAvail) {
    tibAvail.textContent = (tokenState.available ?? 0).toLocaleString();
    tibAvail.className   = 'tib-val' + (tokenState.blocked ? ' blocked' : tokenState.warn ? ' warn' : ' available');
  }

  const blockedNotice = document.getElementById('quota-blocked-notice');
  const warnToast     = document.getElementById('quota-warn-toast');
  const sendBtn       = document.getElementById('send-btn');

  if (tokenState.blocked) {
    if (blockedNotice) blockedNotice.classList.add('visible');
    if (warnToast)     warnToast.classList.remove('visible');
    if (sendBtn && !isLoading) sendBtn.disabled = true;
  } else {
    if (blockedNotice) blockedNotice.classList.remove('visible');
    if (sendBtn && !isLoading) sendBtn.disabled = false;
    if (warnToast) {
      if (tokenState.warn) {
        const wt = document.getElementById('quota-warn-text');
        if (wt) wt.textContent = `Low quota: ${fmtTokens(tokenState.available)} remaining. Contact admin.`;
        warnToast.classList.add('visible');
      } else {
        warnToast.classList.remove('visible');
      }
    }
  }
}

   STREAMING SEND
═══════════════════════════════════════════════════════════════ */
async function sendQuery() {
  const input    = document.getElementById('query-input');
  const question = input.value.trim();
  if (!question || isLoading) return;

  if (tokenState.blocked) {
    const n = document.getElementById('quota-blocked-notice');
    if (n) { n.classList.add('visible'); n.scrollIntoView({ behavior: 'smooth', block: 'nearest' }); }
    return;
  }
  if (_quotaState.blocked && !_quotaState.unlimited) {
    const n = document.getElementById('quota-blocked-notice');
    if (n) { n.classList.add('visible'); n.scrollIntoView({ behavior: 'smooth', block: 'nearest' }); }
    return;
  }

  if (!currentSession) newChat();
  const welcome = document.getElementById('welcome-screen');
  if (welcome) welcome.remove();

  if (currentSession.messages.length === 0) {
    currentSession.title = question.length > 42 ? question.slice(0,42) + '…' : question;
    document.getElementById('session-title').textContent = currentSession.title;
    renderHistory();
  }

  const userTs = nowTs();
  currentSession.messages.push({ role: 'user', content: question, ts: userTs });
  input.value = ''; autoResize(input);

  const area = document.getElementById('chat-area');
  area.appendChild(buildUserRow(question, userTs));
  scrollToBottom();

  isLoading = true;
  document.getElementById('send-btn').disabled = true;

  const msgId = Date.now();
  const aiRow = document.createElement('div');
  aiRow.className = 'msg-row ai'; aiRow.dataset.msgId = msgId;
  const aiAv = document.createElement('div');
  aiAv.className = 'avatar ai'; aiAv.textContent = 'S';
  const bw = document.createElement('div');
  bw.className = 'bubble-wrap';
  const typingBubble = document.createElement('div');
  typingBubble.className = 'bubble ai-bubble'; typingBubble.id = `typing-${msgId}`;
  typingBubble.innerHTML = `<div style="display:flex;gap:4px;align-items:center;padding:2px 0"><div class="typing-dot"></div><div class="typing-dot"></div><div class="typing-dot"></div></div>`;
  bw.appendChild(typingBubble);
  bw.appendChild(buildProcessingBar());
  aiRow.appendChild(aiAv); aiRow.appendChild(bw);
  area.appendChild(aiRow); scrollToBottom();

  let thinkingBlock=null, thinkingBody=null, sqlBlock=null, metaBlock=null, chartBlock=null;
  let rowsBadge=null, tsSpan=null, tablesUsed=[], currentSql='', resultData=null, vizData=null, attempt=0;
  const snapshot = {};

  function removeTyping() { const t = document.getElementById(`typing-${msgId}`); if (t) t.remove(); }
  function ensureThinking() {
    if (!thinkingBlock) {
      thinkingBlock = buildThinkingBlock(msgId);
      thinkingBody  = thinkingBlock.querySelector(`#think-${msgId}`);
      removeTyping(); bw.appendChild(thinkingBlock); scrollToBottom();
    }
  }
  function addThinkingStep(iconClass, iconChar, html) {
    ensureThinking();
    const step = document.createElement('div');
    step.className = 'thinking-step';
    step.innerHTML = `<div class="step-icon ${iconClass}">${iconChar}</div><div>${html}</div>`;
    thinkingBody.appendChild(step); scrollToBottom();
  }
  function addOrReplaceTimestamp(tokens) {
    if (tsSpan) tsSpan.remove();
    tsSpan = document.createElement('span'); tsSpan.className = 'msg-ts';
    const timeStr = nowTs();
    if (tokens != null) {
      tsSpan.innerHTML = `<strong style="color:var(--text-primary);font-weight:600;">${tokens.toLocaleString()} tokens</strong><span style="color:var(--text-muted);margin-left:4px;">| ${timeStr}</span>`;
    } else { tsSpan.textContent = timeStr; }
    bw.appendChild(tsSpan);
  }

  let fetchDone = false;
  try {
    const resp = await fetch(`${API_BASE}/query/stream`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ question, max_retries: 3 }),
    });
    if (!resp.ok) throw new Error(`HTTP ${resp.status}`);

    const reader = resp.body.getReader();
    const decoder = new TextDecoder();
    let buffer = '';

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });
      const frames = buffer.split('\n\n');
      buffer = frames.pop();

      for (const frame of frames) {
        if (!frame.trim() || frame.startsWith(':')) continue;
        let eventType = 'message', dataLine = '';
        for (const line of frame.split('\n')) {
          if (line.startsWith('event:')) eventType = line.slice(6).trim();
          if (line.startsWith('data:'))  dataLine  = line.slice(5).trim();
        }
        if (!dataLine) continue;
        let payload;
        try { payload = JSON.parse(dataLine); } catch { continue; }

        switch (eventType) {
          case 'tables_selected': {
            tablesUsed = payload.tables || [];
            const enriched = payload.enriched_query || '';
            addThinkingStep('tables','T',
              `<strong>Tables:</strong> ${escHtml(tablesUsed.join(', '))}<br>
               <span style="color:var(--text-muted)">${escHtml(payload.reasoning||'')}</span>
               ${enriched ? `<div style="margin-top:7px;padding:7px 10px;background:var(--blue-50);border:1px solid var(--blue-100);border-radius:var(--radius-sm);font-size:12px;line-height:1.5;color:var(--blue-800);">
                 <span style="font-size:9.5px;font-weight:600;letter-spacing:0.5px;text-transform:uppercase;color:var(--blue-600);display:block;margin-bottom:3px;">Understood as</span>${escHtml(enriched)}</div>` : ''}`);
            addOrReplaceTimestamp(); updateProcessingBar('sql','⟨⟩','Generating SQL…'); break;
          }
          case 'sql_generated': {
            attempt = payload.attempt; currentSql = payload.sql || '';
            addThinkingStep('sql','⟨⟩', attempt===1 ? `<strong>SQL generated</strong> — attempt ${attempt}` : `<strong>Retrying SQL</strong> — attempt ${attempt}`);
            addOrReplaceTimestamp(); scrollToBottom();
            updateProcessingBar('valid','✔', attempt===1 ? 'Validating SQL…' : `Fixing SQL — attempt ${attempt}…`); break;
          }
          case 'sql_error': {
            const notice = document.createElement('div');
            notice.className = 'retry-notice';
            notice.innerHTML = `<svg viewBox="0 0 24 24" width="13" height="13" fill="currentColor" style="color:var(--text-muted)"><path d="M17.65 6.35C16.2 4.9 14.21 4 12 4c-4.42 0-7.99 3.58-7.99 8s3.57 8 7.99 8c3.73 0 6.84-2.55 7.73-6h-2.08c-.82 2.33-3.04 4-5.65 4-3.31 0-6-2.69-6-6s2.69-6 6-6c1.66 0 3.14.69 4.22 1.78L13 11h7V4l-2.35 2.35z"/></svg>Validation failed — retrying (attempt ${payload.attempt+1})`;
            if (tsSpan && tsSpan.parentNode===bw) bw.insertBefore(notice,tsSpan); else bw.appendChild(notice);
            addOrReplaceTimestamp(); scrollToBottom();
            updateProcessingBar('sql','⟨⟩',`Retrying SQL — attempt ${payload.attempt+1}…`); break;
          }
          case 'result': {
            resultData = payload; tablesUsed = payload.tables_used || tablesUsed;
            addThinkingStep('done','✔',`<strong>Completed</strong> in ${payload.attempts} attempt${payload.attempts>1?'s':''} · ${payload.rows} row${payload.rows!==1?'s':''} returned`);
            const resultFooter = document.createElement('div');
            resultFooter.style.cssText = 'display:flex;align-items:center;gap:8px;flex-wrap:wrap;margin-top:4px;';
            rowsBadge = buildRowsBadge(payload.rows,(payload.columns||[]).length);
            resultFooter.appendChild(rowsBadge);
            if (payload.output_csv) {
              const csvFile = payload.output_csv.split('/').pop();
              const csvBtn = document.createElement('a');
              csvBtn.className = 'csv-download-btn';
              csvBtn.href = `${API_BASE}/download/${encodeURIComponent(csvFile)}`;
              csvBtn.download = csvFile;
              csvBtn.innerHTML = `<svg viewBox="0 0 24 24" width="12" height="12" fill="currentColor"><path d="M19 9h-4V3H9v6H5l7 7 7-7zM5 18v2h14v-2H5z"/></svg> Download CSV`;
              resultFooter.appendChild(csvBtn);
              const dtBtn = document.createElement('button');
              dtBtn.className = 'csv-download-btn';
              dtBtn.style.cssText += ';background:var(--lavender-50);color:var(--lavender-600);border-color:var(--lavender-200);cursor:pointer;';
              dtBtn.innerHTML = `<svg viewBox="0 0 24 24" width="12" height="12" fill="currentColor"><path d="M3 13h2v-2H3v2zm0 4h2v-2H3v2zm0-8h2V7H3v2zm4 4h14v-2H7v2zm0 4h14v-2H7v2zM7 7v2h14V7H7z"/></svg> View Table`;
              dtBtn._csvUrl   = `${API_BASE}/download/${encodeURIComponent(csvFile)}`;
              dtBtn._csvTitle = resultData ? (resultData.tables_used||[]).join(', ') || csvFile : csvFile;
              dtBtn.onclick = () => { if (window.SarthiDatatable) SarthiDatatable.open(dtBtn._csvUrl, dtBtn._csvTitle); };
              resultFooter.appendChild(dtBtn);
              snapshot._dtBtn = dtBtn;
            }
            if (tsSpan && tsSpan.parentNode===bw) bw.insertBefore(resultFooter,tsSpan); else bw.appendChild(resultFooter);
            snapshot.thinking = thinkingBlock; snapshot.rowsBadge = resultFooter;
            addOrReplaceTimestamp(); scrollToBottom(); updateProcessingBar('viz','V','Deciding chart type…'); break;
          }
          case 'viz_meta': {
            vizData = payload;
            if (metaBlock) metaBlock.remove();
            metaBlock = buildMetaGrid(tablesUsed, payload);
            const insertAfter = thinkingBlock || sqlBlock;
            if (insertAfter && insertAfter.nextSibling) bw.insertBefore(metaBlock, insertAfter.nextSibling);
            else if (tsSpan && tsSpan.parentNode===bw) bw.insertBefore(metaBlock,tsSpan);
            else bw.appendChild(metaBlock);
            addThinkingStep('viz','V',`<strong>Chart decision:</strong> ${escHtml(payload.chart_type)} — ${escHtml(payload.reasoning||'')}`);
            snapshot.meta = metaBlock;
            addOrReplaceTimestamp(); scrollToBottom(); updateProcessingBar('chart','◎','Generating chart…'); break;
          }
          case 'chart_ready': {
            if (payload.chart_type_changed && payload.actual_chart_type && metaBlock) {
              const cp = metaBlock.querySelector('.meta-pill.chart');
              if (cp) cp.textContent = payload.actual_chart_type;
              if (vizData) vizData.chart_type = payload.actual_chart_type;
            }
            if (payload.output_html) {
              const chartFile = payload.output_html.split('/').pop();
              chartBlock = buildChartFrame(chartFile, vizData?.title || 'Chart');
              if (tsSpan && tsSpan.parentNode===bw) bw.insertBefore(chartBlock,tsSpan); else bw.appendChild(chartBlock);
              snapshot.chart = chartBlock;
            } else if (payload.error) {
              bw.appendChild(buildErrorEl(`Chart generation failed: ${payload.error}`));
            }
            addOrReplaceTimestamp(); scrollToBottom(); break;
          }
          case 'tokens_adjusted': {
            if (payload.balance !== undefined) {
              tokenState.available = payload.balance;
              tokenState.warn    = payload.balance <= 25000 && payload.balance > 0;
              tokenState.blocked = payload.balance <= 0;
              updateTokenUI();
            }
            break;
          }
          case 'tokens_refunded': {
            if (payload.refunded) { tokenState.available = (tokenState.available||0) + payload.refunded; updateTokenUI(); }
            break;
          }
          case 'tokens_reserved': { break; }
          case 'pipeline_error': {
            removeTyping();
            const errEl = buildErrorEl(payload.error || 'Something went wrong.');
            if (tsSpan && tsSpan.parentNode===bw) bw.insertBefore(errEl,tsSpan); else bw.appendChild(errEl);
            addOrReplaceTimestamp();
            currentSession.messages.push({ role:'ai', type:'error', content: payload.error||'Something went wrong.' });
            removeProcessingBar(); scrollToBottom(); break;
          }
          case 'done': {
            removeProcessingBar(); fetchDone = true;
            const totalTokens = payload.total_tokens ?? null;
            addOrReplaceTimestamp(totalTokens);
            fetchQuotaBalance(); // sync token UI with DB
            if (resultData) {
              currentSession.messages.push({ role:'ai', type:'result', snapshot: { thinking:thinkingBlock, meta:metaBlock, sql:sqlBlock, chart:chartBlock, rowsBadge } });
            }
            break;
          }
        }
      }
    }
  } catch (err) {
    removeProcessingBar(); removeTyping();
    bw.appendChild(buildErrorEl(`Could not reach the server. Is the API running at ${API_BASE}?`));
    bw.appendChild(tsEl());
    currentSession.messages.push({ role:'ai', type:'error', content: err.message });
  }

  isLoading = false;
  document.getElementById('send-btn').disabled = false;
  scrollToBottom();
}

function toggleThinking(id) {
  const body = document.getElementById(id);
  const chev = document.getElementById('chev-' + id);
  if (!body) return;
  body.classList.toggle('open'); chev.classList.toggle('open');
}

function copySQL(elId) {
  const el = document.getElementById(elId); if (!el) return;
  navigator.clipboard.writeText(el.textContent).catch(()=>{});
  const btn = el.previousElementSibling?.querySelector('.sql-copy-btn');
  if (btn) { btn.textContent = 'Copied!'; setTimeout(() => btn.textContent = 'Copy', 1500); }
}

function scrollToBottom() {
  const area = document.getElementById('chat-area');
  setTimeout(() => area.scrollTop = area.scrollHeight, 30);
}

function handleKey(e) {
  if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendQuery(); }
}

function autoResize(el) {
  el.style.height = 'auto';
  el.style.height = Math.min(el.scrollHeight, 120) + 'px';
}

function escHtml(str) {
  if (!str) return '';
  return String(str).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function openChartModal(chartUrl, title) {
  const overlay = document.getElementById('chart-modal-overlay');
  document.getElementById('chart-modal-iframe').src = chartUrl;
  document.getElementById('chart-modal-title').textContent = title || 'Chart';
  overlay.classList.add('open');
  document.body.style.overflow = 'hidden';
}

function closeChartModal(event) {
  if (event && event.target !== document.getElementById('chart-modal-overlay')) return;
  const overlay = document.getElementById('chart-modal-overlay');
  overlay.classList.remove('open');
  document.getElementById('chart-modal-iframe').src = '';
  document.body.style.overflow = '';
}
