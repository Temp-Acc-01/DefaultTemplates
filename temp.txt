def _extract_tokens(response) -> int:
    """Extract total token count from a LangChain Gemini response."""
    try:
        usage = getattr(response, 'usage_metadata', None)
        if usage:
            return usage.get('total_tokens', 0)
        meta  = getattr(response, 'response_metadata', {})
        usage2 = meta.get('usage_metadata', {})
        if usage2:
            return usage2.get('total_token_count', 0)
    except Exception:
        pass
    return 0

function addOrReplaceTimestamp(tokens) {
  if (tsSpan) tsSpan.remove();
  tsSpan = document.createElement('span');
  tsSpan.className = 'msg-ts';
  const timeStr = nowTs();
  tsSpan.textContent = tokens != null
    ? `${tokens.toLocaleString()} tokens | ${timeStr}`
    : timeStr;
  bw.appendChild(tsSpan);
