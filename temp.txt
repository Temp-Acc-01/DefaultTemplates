-- Token wallets: one row per user, tracks balance
CREATE TABLE IF NOT EXISTS token_wallets (
    user_id BIGINT PRIMARY KEY,  -- matches nlq_user_details.id or username lookup
    total_allocated BIGINT NOT NULL DEFAULT 0,
    total_used BIGINT NOT NULL DEFAULT 0,
    available_tokens BIGINT NOT NULL DEFAULT 0,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES nlq_user_details(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Token transactions: immutable audit ledger
CREATE TABLE IF NOT EXISTS token_transactions (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT NOT NULL,
    type ENUM('ALLOCATE', 'RESERVE', 'DEDUCT', 'REFUND', 'ADJUST') NOT NULL,
    tokens BIGINT NOT NULL,
    request_id VARCHAR(128),          -- unique per API call (e.g., UUID)
    model_name VARCHAR(64),           -- e.g., "gemini-2.5-pro"
    balance_before BIGINT,
    balance_after BIGINT,
    metadata JSON,                    -- e.g., {"query": "top 5 customers", "attempt": 2}
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_user (user_id),
    INDEX idx_request (request_id),
    INDEX idx_created (created_at),
    FOREIGN KEY (user_id) REFERENCES nlq_user_details(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

"""
token_manager.py -- Production Token Wallet System

Design:
- token_wallets: current balance per user (atomic updates)
- token_transactions: immutable audit ledger
- Reserve → Deduct → Refund flow for LLM queries
- Atomic conditional UPDATE prevents race conditions
"""

# Thresholds
WARN_THRESHOLD     = 25_000   # show warning when available <= this
CRITICAL_THRESHOLD = 0        # block when available <= this
DEFAULT_RESERVE    = 25_000   # reserve this much per query by default
MAX_RESERVE        = 100_000  # hard cap on reserve amount

# DB Helpers

def _get_conn():
    """Get a new MySQL connection."""
    return pymysql.connect(
        host=MYSQL_HOST,
        port=MYSQL_PORT,
        user=MYSQL_USER,
        password=MYSQL_PASS,
        database=MYSQL_DB,
        cursorclass=pymysql.cursors.DictCursor,
        connect_timeout=10,
        autocommit=False,  # we manage transactions manually
    )


def _get_username() -> Optional[str]:
    """Get current username from Flask session."""
    return session.get("username")


def _ensure_wallet(username: str, conn) -> None:
    """Create wallet row if it doesn't exist (idempotent)."""
    with conn.cursor() as cur:
        cur.execute(f"""
            INSERT INTO {WALLET_TABLE} (user_id, total_allocated, total_used, available_tokens)
            SELECT %s, 0, 0, 0
            WHERE NOT EXISTS (
                SELECT 1 FROM {WALLET_TABLE} WHERE user_id = %s
            )
        """, (username, username))
    # Do NOT commit here — caller manages transaction


def _log_transaction(
    conn,
    username: str,
    txn_type: str,
    tokens: int,
    request_id: Optional[str] = None,
    model_name: Optional[str] = None,
    balance_before: Optional[int] = None,
    balance_after: Optional[int] = None,
    metadata: Optional[Dict] = None,
) -> None:
    """Insert an immutable transaction record."""
    with conn.cursor() as cur:
        cur.execute(f"""
            INSERT INTO {TXN_TABLE} (
                user_id, type, tokens, request_id, model_name,
                balance_before, balance_after, metadata
            ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
        """, (
            username, txn_type, tokens, request_id, model_name,
            balance_before, balance_after, metadata
        ))


def get_wallet(username: str) -> Dict[str, Any]:
    """
    Get current wallet state for a user.
    Returns: {available, total_allocated, total_used, warn, blocked}
    """
    conn = _get_conn()
    try:
        with conn.cursor() as cur:
            cur.execute(f"""
                SELECT available_tokens, total_allocated, total_used
                FROM {WALLET_TABLE}
                WHERE user_id = %s LIMIT 1
            """, (username,))
            row = cur.fetchone()
        
        if not row:
            # Wallet doesn't exist yet — treat as zero balance
            return {
                "available": 0,
                "total_allocated": 0,
                "total_used": 0,
                "warn": False,
                "blocked": True,
            }
        
        available = row["available_tokens"] or 0
        allocated = row["total_allocated"] or 0
        used = row["total_used"] or 0
        
        return {
            "available": available,
            "total_allocated": allocated,
            "total_used": used,
            "warn": 0 < available <= WARN_THRESHOLD,
            "blocked": available <= CRITICAL_THRESHOLD,
        }
    finally:
        conn.close()


def allocate_tokens(username: str, amount: int, request_id: Optional[str] = None) -> Dict[str, Any]:
    """
    Admin operation: add tokens to a user's wallet.
    Atomic: updates wallet + logs transaction in single transaction.
    """
    if amount <= 0:
        return {"success": False, "message": "Amount must be positive"}
    
    conn = _get_conn()
    try:
        with conn:  # transaction context
            _ensure_wallet(username, conn)
            
            # Read current balance
            with conn.cursor() as cur:
                cur.execute(f"""
                    SELECT available_tokens FROM {WALLET_TABLE}
                    WHERE user_id = %s FOR UPDATE
                """, (username,))
                row = cur.fetchone()
                balance_before = row["available_tokens"] if row else 0
            
            # Update wallet
            with conn.cursor() as cur:
                cur.execute(f"""
                    UPDATE {WALLET_TABLE}
                    SET total_allocated = total_allocated + %s,
                        available_tokens = available_tokens + %s,
                        updated_at = NOW()
                    WHERE user_id = %s
                """, (amount, amount, username))
            
            balance_after = (balance_before or 0) + amount
            
            # Log transaction
            _log_transaction(
                conn, username, "ALLOCATE", amount,
                request_id=request_id,
                balance_before=balance_before,
                balance_after=balance_after,
                metadata={"admin_action": True}
            )
        
        return {
            "success": True,
            "available": balance_after,
            "total_allocated": (balance_before or 0) + amount,
            "message": f"Allocated {amount:,} tokens"
        }
    except Exception as e:
        logger.error("allocate_tokens error: %s", e)
        return {"success": False, "message": f"DB error: {e}"}
    finally:
        conn.close()


def reserve_tokens(
    username: str,
    amount: int,
    request_id: str,
    model_name: Optional[str] = None,
    metadata: Optional[Dict] = None,
) -> Dict[str, Any]:
    """
    Reserve tokens for an LLM query (pre-authorization).
    Atomic conditional update: only succeeds if sufficient balance.
    Reserved tokens are still "available" but earmarked — we track via transaction log.
    
    Note: We don't physically move tokens to a "reserved" column.
    Instead, we log a RESERVE transaction and rely on the subsequent
    DEDUCT/REFUND to finalize. This keeps the wallet simple.
    """
    if amount <= 0:
        return {"success": True, "reserved": 0, "message": "No reserve needed"}
    
    conn = _get_conn()
    try:
        with conn:  # transaction
            _ensure_wallet(username, conn)
            
            # Atomic conditional update
            with conn.cursor() as cur:
                affected = cur.execute(f"""
                    UPDATE {WALLET_TABLE}
                    SET available_tokens = available_tokens - %s,
                        updated_at = NOW()
                    WHERE user_id = %s AND available_tokens >= %s
                """, (amount, username, amount))
            
            if affected == 0:
                # Insufficient balance
                current = get_wallet(username)
                return {
                    "success": False,
                    "available": current["available"],
                    "blocked": True,
                    "message": f"Insufficient balance. Need {amount:,}, have {current['available']:,}"
                }
            
            # Read new balance
            with conn.cursor() as cur:
                cur.execute(f"""
                    SELECT available_tokens FROM {WALLET_TABLE}
                    WHERE user_id = %s
                """, (username,))
                row = cur.fetchone()
                balance_after = row["available_tokens"]
            
            balance_before = balance_after + amount
            
            # Log RESERVE transaction
            _log_transaction(
                conn, username, "RESERVE", amount,
                request_id=request_id,
                model_name=model_name,
                balance_before=balance_before,
                balance_after=balance_after,
                metadata=metadata or {}
            )
        
        return {
            "success": True,
            "reserved": amount,
            "available": balance_after,
            "message": "Reserved"
        }
    except Exception as e:
        logger.error("reserve_tokens error: %s", e)
        return {"success": False, "message": f"DB error: {e}"}
    finally:
        conn.close()


def finalize_usage(
    username: str,
    request_id: str,
    actual_used: int,
    reserved_amount: int,
    model_name: Optional[str] = None,
    metadata: Optional[Dict] = None,
) -> Dict[str, Any]:
    """
    After LLM response: deduct actual usage, refund the rest.
    Two atomic operations in one transaction.
    """
    if actual_used < 0:
        return {"success": False, "message": "Invalid token count"}
    
    conn = _get_conn()
    try:
        with conn:  # transaction
            _ensure_wallet(username, conn)
            
            # Read current balance
            with conn.cursor() as cur:
                cur.execute(f"""
                    SELECT available_tokens FROM {WALLET_TABLE}
                    WHERE user_id = %s FOR UPDATE
                """, (username,))
                row = cur.fetchone()
                current_avail = row["available_tokens"] if row else 0
            
            refund = max(0, reserved_amount - actual_used)
            to_deduct = actual_used  # net deduction from reserved pool
            
            # If reserved > actual, we refund the difference back to available
            if refund > 0:
                with conn.cursor() as cur:
                    cur.execute(f"""
                        UPDATE {WALLET_TABLE}
                        SET available_tokens = available_tokens + %s,
                            updated_at = NOW()
                        WHERE user_id = %s
                    """, (refund, username))
                # Log REFUND
                _log_transaction(
                    conn, username, "REFUND", refund,
                    request_id=request_id,
                    model_name=model_name,
                    balance_before=current_avail,
                    balance_after=current_avail + refund,
                    metadata={**(metadata or {}), "reserved": reserved_amount}
                )
                current_avail += refund
            
            # Log the final DEDUCT for actual usage
            _log_transaction(
                conn, username, "DEDUCT", actual_used,
                request_id=request_id,
                model_name=model_name,
                balance_before=current_avail,
                balance_after=current_avail - actual_used,
                metadata={**(metadata or {}), "reserved": reserved_amount}
            )
            
            # Update total_used counter
            with conn.cursor() as cur:
                cur.execute(f"""
                    UPDATE {WALLET_TABLE}
                    SET total_used = total_used + %s,
                        updated_at = NOW()
                    WHERE user_id = %s
                """, (actual_used, username))
        
        new_wallet = get_wallet(username)
        return {
            "success": True,
            "actual_used": actual_used,
            "refunded": refund,
            "available": new_wallet["available"],
            "total_used": new_wallet["total_used"],
            "message": "Finalized"
        }
    except Exception as e:
        logger.error("finalize_usage error: %s", e)
        return {"success": False, "message": f"DB error: {e}"}
    finally:
        conn.close()


def deduct_tokens_immediate(
    username: str,
    amount: int,
    request_id: Optional[str] = None,
    model_name: Optional[str] = None,
    metadata: Optional[Dict] = None,
) -> Dict[str, Any]:
    """
    Simple deduct (no reserve) for non-LLM operations.
    Atomic conditional update.
    """
    if amount <= 0:
        return {"success": True, "message": "No deduction needed"}
    
    conn = _get_conn()
    try:
        with conn:
            _ensure_wallet(username, conn)
            
            with conn.cursor() as cur:
                affected = cur.execute(f"""
                    UPDATE {WALLET_TABLE}
                    SET available_tokens = available_tokens - %s,
                        total_used = total_used + %s,
                        updated_at = NOW()
                    WHERE user_id = %s AND available_tokens >= %s
                """, (amount, amount, username, amount))
            
            if affected == 0:
                current = get_wallet(username)
                return {
                    "success": False,
                    "available": current["available"],
                    "blocked": True,
                    "message": f"Insufficient balance"
                }
            
            # Read new balance
            with conn.cursor() as cur:
                cur.execute(f"""
                    SELECT available_tokens, total_used FROM {WALLET_TABLE}
                    WHERE user_id = %s
                """, (username,))
                row = cur.fetchone()
            
            _log_transaction(
                conn, username, "DEDUCT", amount,
                request_id=request_id,
                model_name=model_name,
                balance_before=(row["available_tokens"] or 0) + amount,
                balance_after=row["available_tokens"],
                metadata=metadata or {}
            )
        
        return {
            "success": True,
            "available": row["available_tokens"],
            "total_used": row["total_used"],
            "message": "Deducted"
        }
    except Exception as e:
        logger.error("deduct_tokens_immediate error: %s", e)
        return {"success": False, "message": f"DB error: {e}"}
    finally:
        conn.close()


def get_transaction_history(username: str, limit: int = 50) -> list:
    """Get recent transactions for a user (for audit/debug)."""
    conn = _get_conn()
    try:
        with conn.cursor() as cur:
            cur.execute(f"""
                SELECT id, type, tokens, request_id, model_name,
                       balance_before, balance_after, metadata, created_at
                FROM {TXN_TABLE}
                WHERE user_id = %s
                ORDER BY created_at DESC
                LIMIT %s
            """, (username, limit))
            return cur.fetchall()
    finally:
        conn.close()

tokens_bp = Blueprint("tokens", __name__)


def token_login_required(f):
    """Decorator: require authenticated session."""
    @functools.wraps(f)
    def decorated(*args, **kwargs):
        if not _get_username():
            return jsonify({"status": "error", "message": "Not authenticated"}), 401
        return f(*args, **kwargs)
    return decorated


@tokens_bp.route("/tokens/balance", methods=["GET"])
@token_login_required
def balance_route():
    """GET /tokens/balance — current wallet state."""
    username = _get_username()
    wallet = get_wallet(username)
    return jsonify({"status": "success", **wallet})


@tokens_bp.route("/tokens/reserve", methods=["POST"])
@token_login_required
def reserve_route():
    """
    POST /tokens/reserve
    Body: { "amount": 25000, "request_id": "uuid", "model_name": "gemini-2.5-pro" }
    Reserve tokens before calling LLM.
    """
    username = _get_username()
    body = request.get_json(silent=True) or {}
    
    result = reserve_tokens(
        username=username,
        amount=int(body.get("amount") or DEFAULT_RESERVE),
        request_id=body.get("request_id") or str(uuid.uuid4()),
        model_name=body.get("model_name"),
        metadata=body.get("metadata"),
    )
    
    status = 200 if result["success"] else (402 if result.get("blocked") else 400)
    return jsonify({"status": "success" if result["success"] else "error", **result}), status


@tokens_bp.route("/tokens/finalize", methods=["POST"])
@token_login_required
def finalize_route():
    """
    POST /tokens/finalize
    Body: { "request_id": "uuid", "actual_used": 3200, "reserved_amount": 25000 }
    Call after LLM response to deduct actual + refund remainder.
    """
    username = _get_username()
    body = request.get_json(silent=True) or {}
    
    result = finalize_usage(
        username=username,
        request_id=body.get("request_id"),
        actual_used=int(body.get("actual_used") or 0),
        reserved_amount=int(body.get("reserved_amount") or 0),
        model_name=body.get("model_name"),
        metadata=body.get("metadata"),
    )
    
    status = 200 if result["success"] else 400
    return jsonify({"status": "success" if result["success"] else "error", **result}), status


@tokens_bp.route("/tokens/history", methods=["GET"])
@token_login_required
def history_route():
    """GET /tokens/history — recent transactions for audit."""
    username = _get_username()
    limit = min(int(request.args.get("limit") or 50), 200)
    txns = get_transaction_history(username, limit)
    return jsonify({"status": "success", "transactions": txns})


# Admin-only route: allocate tokens to any user
@tokens_bp.route("/tokens/admin/allocate", methods=["POST"])
@token_login_required
def admin_allocate_route():
    """
    POST /tokens/admin/allocate
    Body: { "target_user": "username", "amount": 100000 }
    Requires: session user must be in ADMIN_USERS list (configurable).
    """
    ADMIN_USERS = os.environ.get("TOKEN_ADMIN_USERS", "admin,superuser").split(",")
    current_user = _get_username()
    
    if current_user not in ADMIN_USERS:
        return jsonify({"status": "error", "message": "Admin access required"}), 403
    
    body = request.get_json(silent=True) or {}
    target = body.get("target_user", "").strip()
    amount = int(body.get("amount") or 0)
    
    if not target or amount <= 0:
        return jsonify({"status": "error", "message": "target_user and positive amount required"}), 400
    
    result = allocate_tokens(
        username=target,
        amount=amount,
        request_id=f"admin_{current_user}_{uuid.uuid4().hex[:8]}",
    )
    
    status = 200 if result["success"] else 400
    return jsonify({"status": "success" if result["success"] else "error", **result}), status
