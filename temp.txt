@dataclass
class VizMeta:
    chart_type:      str
    title:           str
    x_axis:          Optional[str] = None
    x_axis_label:    Optional[str] = None    # ← ADD
    y_axis:          Optional[str] = None
    y_axis_label:    Optional[str] = None    # ← ADD
    y_columns:       list          = field(default_factory=list)
    secondary_axis:  Optional[str] = None
    series_label:    Optional[str] = None
    reasoning:       str           = ""



----------------


CHART_TYPE: <one of: bar, line, pie, scatter, area, heatmap, table>
TITLE: <concise chart title, max 10 words>
X_AXIS: <exact column name for primary/category axis. MANDATORY for bar/line/area/scatter.>
X_AXIS_LABEL: <human-readable label for X axis — strip underscores, humanize abbreviations.
               Examples: DistrictName→District, OrderDate→Date, ULB_ID→Urban Local Body.
               For pie: the slice category label. For scatter: horizontal axis label.>
Y_AXIS: <exact column name for primary value axis. MANDATORY for bar/line/area/scatter.>
Y_AXIS_LABEL: <ONE collective human-readable label for what Y measures.
               For multi-series (L0,L1...): derive collective name e.g. "Maturity Level Count".
               For single series: humanize column name e.g. total_orders→Total Orders.
               For pie: the numeric value label e.g. Count, Amount.
               NEVER use raw column names like L0, cnt, amt_sum here.>
Y_COLUMNS: <comma-separated list of ALL numeric column names to plot as series.
            For single-series charts: same as Y_AXIS.
            For multi-series (grouped/stacked bar, multi-line): ALL numeric cols.
            Example: L0_count,L1_count,L2_count,L3_count,L4_count,L5_count>
SECONDARY_AXIS: <column name for secondary axis, or NONE>
SERIES_LABEL: <for multi-series bar: the human-readable label prefix for the series
               e.g. "Maturity Level" or "Category". NONE for single-series.>
REASONING: <3-4 sentences explaining chart choice and series mapping.>

function buildMetaGrid(tablesUsed, viz) {
  const wrap = document.createElement('div');
  wrap.className = 'meta-grid';

  // Use human-readable labels if available, fallback to raw column name
  const xDisplay = viz.x_axis_label || viz.x_axis || '—';
  const yDisplay = viz.y_axis_label || viz.y_axis || '—';

  wrap.innerHTML = `
    <div class="meta-card">
      <div class="meta-label">Chart type</div>
      <div class="meta-value"><span class="meta-pill chart">${escHtml(viz.chart_type)}</span></div>
    </div>
    <div class="meta-card">
      <div class="meta-label">Tables used</div>
      <div class="meta-value">${tablesUsed.map(t => `<span class="meta-pill table-pill">${escHtml(t)}</span>`).join('')}</div>
    </div>
    <div class="meta-card">
      <div class="meta-label">X axis</div>
      <div class="meta-value">${escHtml(xDisplay)}</div>
    </div>
    <div class="meta-card">
      <div class="meta-label">Y axis</div>
      <div class="meta-value">${escHtml(yDisplay)}</div>
    </div>`;
  return wrap;
}
