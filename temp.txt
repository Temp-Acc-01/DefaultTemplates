case 'tokens_adjusted': {
  // Update token state with actual balance from backend
  if (payload.balance !== undefined) {
    tokenState.available = payload.balance;
    tokenState.warn = payload.balance <= 25000 && payload.balance > 0;
    tokenState.blocked = payload.balance <= 0;
    
    // Update UI immediately
    updateTokenUI();
    
    // Show refund message if applicable
    const refundMsg = payload.refunded > 0 
      ? `✓ Refunded ${payload.refunded.toLocaleString()} unused tokens` 
      : `✓ Used ${payload.actual_used.toLocaleString()} tokens`;
    
    const notice = document.createElement('div');
    notice.className = 'retry-notice';
    notice.style.cssText = 'background:#E8F5E9;color:#2E7D32;border-color:#A5D6A7;';
    notice.innerHTML = refundMsg;
    bw.appendChild(notice);
    scrollToBottom();
  }
  break;
}
