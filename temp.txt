# manage_user_tokens.py
import pymysql
import os
from dotenv import load_dotenv
from token_manager import allocate_tokens, _get_conn

load_dotenv()

# ── CONFIG ──
USERNAME = "existing_user"      # Change to target user
PASSWORD = "SecurePass123!"     # Only used if creating a NEW user
TOKENS_TO_ADD = 500_000         # Tokens to ADD (top-up amount)

conn = _get_conn()
try:
    with conn.cursor() as cur:
        # 1. Check if user already exists
        cur.execute("SELECT id FROM nlq_user_details WHERE username = %s", (USERNAME,))
        user_row = cur.fetchone()

        if user_row:
            user_id = user_row['id']
            print(f"✅ User '{USERNAME}' found (ID: {user_id})")
        else:
            # Create new user
            cur.execute(
                "INSERT INTO nlq_user_details (username, password) VALUES (%s, %s)",
                (USERNAME, PASSWORD)
            )
            user_id = cur.lastrowid
            conn.commit()
            print(f"✅ User '{USERNAME}' created (ID: {user_id})")

        # 2. Allocate tokens (works for both new & existing users)
        result = allocate_tokens(user_id, TOKENS_TO_ADD, request_id="manual_topup")
        print(f"✅ Added {TOKENS_TO_ADD:,} tokens to '{USERNAME}'")
        print(f"   New balance: {result['wallet']['available_tokens']:,}")
        print(f"   Total allocated: {result['wallet']['total_allocated']:,}")

except Exception as e:
    print(f"❌ Failed: {e}")
    conn.rollback()
finally:
    conn.close()
