function buildChartFrame(chartFile, title) {
  const wrap = document.createElement('div');
  wrap.className = 'chart-frame-wrap';

  const header = document.createElement('div');
  header.className = 'chart-frame-header';
  header.innerHTML = `
    <span class="chart-frame-title"></span>
    <button class="chart-open-btn"
      onclick="window.open('${API_BASE}/chart/${encodeURIComponent(chartFile)}','_blank')">
      <svg viewBox="0 0 24 24" width="12" height="12" fill="currentColor">
        <path d="M19 19H5V5h7V3H5a2 2 0 00-2 2v14a2 2 0 002 2h14a2 2 0 002-2v-7h-2v7zM14 3v2h3.59l-9.83 9.83 1.41 1.41L19 6.41V10h2V3h-7z"/>
      </svg>
      Open full chart
    </button>`;

  const iframe = document.createElement('iframe');
  iframe.className = 'chart-iframe';
  iframe.src = `${API_BASE}/chart/${encodeURIComponent(chartFile)}`;
  iframe.loading = 'lazy';

  // ── Kill scrollbar once iframe loads ──────────────────────────
  iframe.onload = () => {
    try {
      const idoc = iframe.contentDocument || iframe.contentWindow.document;
      // Inject a style tag that nukes overflow on html and body
      const style = idoc.createElement('style');
      style.textContent = `
        html, body {
          margin: 0 !important;
          padding: 0 !important;
          overflow: hidden !important;
          background: transparent !important;
        }
        canvas { display: block !important; }
      `;
      idoc.head.appendChild(style);

      // Auto-fit iframe height to content so nothing clips or scrolls
      const h = idoc.body.scrollHeight;
      if (h && h > 100) iframe.style.height = h + 'px';
    } catch {
      // cross-origin — can't access, leave fixed height
    }
  };

  wrap.appendChild(header);
  wrap.appendChild(iframe);
  return wrap;
}

function buildProcessingBar() {
  const bar = document.createElement('div');
  bar.className = 'processing-bar';
  bar.id = 'proc-bar-active';  // only one at a time
  bar.innerHTML = `
    <div class="proc-dots">
      <div class="proc-dot"></div>
      <div class="proc-dot"></div>
      <div class="proc-dot"></div>
    </div>
    <div class="proc-stage-icon tables" id="proc-icon">T</div>
    <span id="proc-text">Selecting relevant tables…</span>
  `;
  return bar;
}

function updateProcessingBar(iconClass, iconChar, text) {
  const icon = document.getElementById('proc-icon');
  const txt  = document.getElementById('proc-text');
  if (!icon || !txt) return;
  icon.className = `proc-stage-icon ${iconClass}`;
  icon.textContent = iconChar;
  txt.textContent  = text;
}

function removeProcessingBar() {
  const bar = document.getElementById('proc-bar-active');
  if (!bar) return;
  bar.classList.add('done');
  setTimeout(() => bar.remove(), 320);  // after fade-out
}
