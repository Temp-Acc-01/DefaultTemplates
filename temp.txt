CREATE TABLE IF NOT EXISTS token_wallets (
    user_id INT PRIMARY KEY,  -- ← CHANGE: INT matches nlq_user_details.id
    total_allocated BIGINT NOT NULL DEFAULT 0,
    total_used BIGINT NOT NULL DEFAULT 0,
    available_tokens BIGINT NOT NULL DEFAULT 0,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES nlq_user_details(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS token_transactions (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,  -- ← CHANGE: INT matches nlq_user_details.id
    type ENUM('ALLOCATE', 'RESERVE', 'DEDUCT', 'REFUND', 'ADJUST') NOT NULL,
    tokens BIGINT NOT NULL,
    request_id VARCHAR(128),
    model_name VARCHAR(64),
    balance_before BIGINT,
    balance_after BIGINT,
    metadata JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_user (user_id),
    INDEX idx_request (request_id),
    INDEX idx_created (created_at),
    FOREIGN KEY (user_id) REFERENCES nlq_user_details(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
